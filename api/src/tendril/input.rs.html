19pub(crate) fn parse_input_definition(input: &str) -> Result<RunInputPayload, TendrilError> {
20    if let Some(offset) = top_level_semicolon_offset(input) {
21        return Err(dsl_error(
22            "unexpected `;`; the DSL separator is `,`",
23            None,
24            None,
25            Some("parse"),
26        )
27        .with_detail_entry("offset", json!(offset)));
28    }
29
30    if input.contains('(') {
31        return Ok(RunInputPayload::Actions {
32            actions: parse_dsl_sequence(input)?,
33        });
34    }
35
36    if contains_top_level_comma(input) && looks_like_bare_key_sequence(input) {
37        if let Ok(actions) = parse_dsl_sequence(input) {
38            return Ok(RunInputPayload::Actions { actions });
39        }
40    }
41
42    if let Some(error) = ambiguous_single_dsl_like_input_error(input) {
43        return Err(error);
44    }
45
46    Ok(RunInputPayload::Text {
47        text: input.to_owned(),
48    })
49}

51pub(crate) fn parse_dsl_sequence(sequence: &str) -> Result<Vec<InputAction>, TendrilError> {
52    let trimmed = sequence.trim();
53    if trimmed.is_empty() {
54        return Err(dsl_error(
55            "dsl sequence cannot be empty",
56            None,
57            None,
58            Some("parse"),
59        ));
60    }
61
62    let segments = split_top_level(trimmed)?;
63    let mut actions = Vec::with_capacity(segments.len());
64    for (action_index, segment) in segments.iter().enumerate() {
65        let action = parse_action_segment(segment, action_index)?;
66        actions.push(action);
67    }
68
69    Ok(actions)
70}

88pub(crate) fn execute_run(
89    input: &RunInput,
90    adapter: &dyn PlatformAdapter,
91) -> Result<RunOutput, TendrilError> {
92    let target = resolve_target(input, adapter)?;
93    // Probe adapter-level input support first so platform-specific diagnostics
94    // (for example the Wayland missing-backend error that names `ydotool` and
95    // `wtype`) surface before the generic per-target capability check, which
96    // would otherwise mask the actionable remediation guidance with
97    // `input_not_supported_for_target` (bd-da01d3).
98    adapter.input_support().map_err(TendrilError::from)?;
99    ensure_input_supported(&target)?;
100
101    let adapter_info = adapter.info();
102    let (text, actions) = normalize_payload(&input.payload);
103    let actions = resolve_element_actions(adapter, &input.target, &target, actions)?;
104    validate_actions_for_target(&target, &actions)?;
105    reject_unsafe_browser_navigation_chord(&adapter_info, &target, &actions)?;
106
107    let outcome = adapter.execute_input(&PlatformInputRequest {
108        target_id: target.id.clone(),
109        target: target.kind,
110        target_name: target.name.clone(),
111        bounds: target.bounds.clone(),
112        app_name: target.app_name.clone(),
113        process_id: target.process_id,
114        restore_focus: input.restore_focus,
115        text,
116        actions: actions.clone(),
117    })?;
118
119    Ok(RunOutput {
120        adapter: adapter_info,
121        target: input.target.clone(),
122        focus_required: outcome.focus_required,
123        focus_transferred: outcome.focus_transferred,
124        action_count: outcome.action_count,
125        focused_target: outcome.focused_target,
126        previous_focus: outcome.previous_focus,
127        focus_restored: outcome.focus_restored,
128        pointer_restored: outcome.pointer_restored,
129        restore_error: outcome.restore_error,
130        notes: outcome.notes,
131        execution_lock: None,
132    })
133}

221fn reject_unsafe_browser_navigation_chord(
222    adapter_info: &AdapterInfo,
223    target: &PlatformTargetDescriptor,
224    actions: &[InputAction],
225) -> Result<(), TendrilError> {
226    if adapter_info.platform != PlatformKind::Linux
227        || adapter_info.session != DesktopSession::X11
228        || target.kind != CaptureTargetKind::Window
229        || !looks_like_browser_target(target)
230    {
231        return Ok(());
232    }
233
234    let Some((send_action_index, navigation_text)) = vulnerable_ctrl_l_navigation(actions) else {
235        return Ok(());
236    };
237
238    Err(TendrilError::validation(format!(
239        "refusing X11 browser navigation through Ctrl+L followed by navigation text `{}` because synthetic browser-chrome shortcuts can stay in the focused page control (observed with Firefox page inputs); use capture -> click the visible address bar -> Ctrl+A -> send URL/path -> Return -> recapture/verify instead",
240        summarize_navigation_text(navigation_text)
241    ))
242    .with_code("invalid_run_input")
243    .with_field("input_definition")
244    .with_detail_entry("stage", json!("browser_navigation_preflight"))
245    .with_detail_entry("pattern", json!("x11_browser_ctrl_l_url_send"))
246    .with_detail_entry("target_id", json!(target.id))
247    .with_detail_entry("target_name", json!(target.name))
248    .with_detail_entry("target_app_name", json!(target.app_name))
249    .with_detail_entry("target_title", json!(target.title))
250    .with_detail_entry("action_index", json!(send_action_index))
251    .with_detail_entry("url_preview", json!(summarize_navigation_text(navigation_text)))
252    .with_detail_entry(
253        "navigation_text_preview",
254        json!(summarize_navigation_text(navigation_text)),
255    )
256    .with_detail_entry(
257        "remediation",
258        json!("Do not rely on Ctrl+L/Cmd+L for browser navigation on X11 Firefox when a page input may already be focused. Capture the browser, click the visible address bar coordinates, run hold(ctrl),a,release(ctrl),send(\"URL_OR_PATH\"),Return, then recapture and verify the page changed before continuing."),
259    ))
260}

286fn vulnerable_ctrl_l_navigation(actions: &[InputAction]) -> Option<(usize, &str)> {
287    let significant_actions = actions
288        .iter()
289        .enumerate()
290        .filter(|(_, action)| !matches!(action, InputAction::Wait { .. }))
291        .collect::<Vec<_>>();
292    if significant_actions.len() < 3 {
293        return None;
294    }
295
296    for start in 0..=significant_actions.len() - 3 {
297        let (_, first) = significant_actions[start];
298        let (_, second) = significant_actions[start + 1];
299        let (_, third) = significant_actions[start + 2];
300        if !(is_ctrl_hold(first) && is_key(second, "l") && is_ctrl_release(third)) {
301            continue;
302        }
303
304        let mut pending_url_send = None;
305        for (action_index, action) in significant_actions.iter().skip(start + 3).copied() {
306            match action {
307                InputAction::Click { .. }
308                | InputAction::ElementClick { .. }
309                | InputAction::DoubleClick { .. }
310                | InputAction::Drag { .. }
311                | InputAction::Scroll { .. } => break,
312                InputAction::Send { text } if looks_like_navigation_text(text) => {
313                    pending_url_send = Some((action_index, text.as_str()));
314                }
315                _ if is_return_key(action) => {
316                    if let Some((send_index, navigation_text)) = pending_url_send {
317                        return Some((send_index, navigation_text));
318                    }
319                }
320                _ => {}
321            }
322        }
323    }
324
325    None
326}

354fn looks_like_navigation_text(text: &str) -> bool {
355    let trimmed = text.trim();
356    let lower = trimmed.to_ascii_lowercase();

358        || lower.starts_with("about:")
359        || lower.starts_with("chrome:")

367    // Linux/X11 browsers accept absolute POSIX paths such as `/tmp/file.txt`
368    // in the address bar and navigate to them as local filesystem targets.

371        || is_absolute_windows_unc_path(text)

378fn is_absolute_windows_drive_path(text: &str) -> bool {
379    let bytes = text.as_bytes();
380    bytes.len() >= 3
381        && bytes[0].is_ascii_alphabetic()
382        && bytes[1] == b':'
383        && matches!(bytes[2], b'\\' | b'/')
384}

386fn is_absolute_windows_unc_path(text: &str) -> bool {
387    let bytes = text.as_bytes();
388    if bytes.len() < 5 || !matches!(bytes[0], b'\\' | b'/') || !matches!(bytes[1], b'\\' | b'/') {
389        return false;
390    }
391
392    let mut components = text[2..]
393        .split(['\\', '/'])
394        .filter(|component| !component.is_empty());
395    components.next().is_some() && components.next().is_some()
396}

408    adapter: &dyn PlatformAdapter,
409    selector: &TargetSelector,
410    target: &PlatformTargetDescriptor,

432        })
433        .collect()
434}

436fn element_click_to_pointer_action(
437    id: &str,
438    target: &PlatformTargetDescriptor,
439    elements: &[ElementDescriptor],
440) -> Result<InputAction, TendrilError> {
441    let element = elements
442        .iter()
443        .find(|element| element.id == id)
444        .ok_or_else(|| {
445            TendrilError::target_not_found("element", id).with_detail_entry(
446                "remediation",
447                json!("Run `tendril --window <id> list-elements --json` again and use one of the returned snapshot-local element ids."),
448            )
449        })?;

451        TendrilError::unsupported_capability(
452            "element_bounds_unavailable",
453            format!(
454                "element `{id}` ({}) did not expose bounds usable for click dispatch",
455                element.name
456            ),
457            Some(json!({
458                "element_id": id,
459                "element_name": element.name,
460                "element_role": element.role,
461            })),
462        )
463    })?;
464
465    let center_x = bounds
466        .x
467        .saturating_add(i32::try_from(bounds.width / 2).unwrap_or(i32::MAX));
468    let center_y = bounds
469        .y
470        .saturating_add(i32::try_from(bounds.height / 2).unwrap_or(i32::MAX));
471    Ok(InputAction::Click {
472        button: MouseButton::Left,
473        x: center_x.saturating_sub(target.bounds.x),
474        y: center_y.saturating_sub(target.bounds.y),
475    })
476}

502    target: &PlatformTargetDescriptor,
503    actions: &[InputAction],
504) -> Result<(), TendrilError> {
505    for (action_index, action) in actions.iter().enumerate() {

508            | InputAction::Hold { .. }
509            | InputAction::Release { .. }
510            | InputAction::Send { .. }
511            | InputAction::Wait { .. }

517                validate_relative_point(*x, *y, &target.bounds, action_index, "pointer_move")?;

519            InputAction::DoubleClick { x, y } => {
520                validate_relative_point(*x, *y, &target.bounds, action_index, "double_click")?;
521            }
522            InputAction::Drag { x0, y0, x1, y1 } => {
523                validate_relative_point(*x0, *y0, &target.bounds, action_index, "drag_start")?;
524                validate_relative_point(*x1, *y1, &target.bounds, action_index, "drag_end")?;
525            }
526            InputAction::Scroll { x, y, .. } => {
527                validate_relative_point(*x, *y, &target.bounds, action_index, "scroll")?;
528            }
529        }
530    }
531
532    Ok(())
533}

559fn top_level_semicolon_offset(input: &str) -> Option<usize> {
560    let mut in_string = false;
561    let mut escape = false;
562    let mut depth = 0_u32;
563
564    for (index, character) in input.char_indices() {
565        if in_string {
566            if escape {
567                escape = false;
568                continue;
569            }
570            match character {
571                '\\' => escape = true,
572                '"' => in_string = false,
573                _ => {}
574            }
575            continue;
576        }
577
578        match character {
579            '"' => in_string = true,
580            '(' => depth = depth.saturating_add(1),
581            ')' => depth = depth.saturating_sub(1),
582            ';' if depth == 0 => return Some(index),
583            _ => {}
584        }
585    }
586
587    None
588}

621fn looks_like_bare_key_sequence(input: &str) -> bool {
622    let mut start = 0_usize;
623    let mut in_string = false;
624    let mut escape = false;
625    let mut depth = 0_i32;
626
627    for (index, character) in input.char_indices() {
628        if in_string {
629            if escape {
630                escape = false;
631                continue;
632            }
633            match character {
634                '\\' => escape = true,
635                '"' => in_string = false,
636                _ => {}
637            }
638            continue;
639        }
640
641        match character {
642            '"' => in_string = true,
643            '(' => depth += 1,
644            ')' => depth -= 1,
645            ',' if depth == 0 => {
646                let raw = &input[start..index];
647                if raw != raw.trim() || parse_key_token(raw).is_none() {
648                    return false;
649                }
650                start = index + character.len_utf8();
651            }
652            _ => {}
653        }
654    }
655
656    let raw = &input[start..];
657    raw == raw.trim() && parse_key_token(raw).is_some()
658}

660fn ambiguous_single_dsl_like_input_error(input: &str) -> Option<TendrilError> {
661    let trimmed = input.trim();
662    if trimmed.is_empty() || trimmed.contains('(') || contains_top_level_comma(trimmed) {
663        return None;
664    }
665
666    if is_known_bare_key_token(trimmed) {
667        return Some(
668            dsl_error(
669                format!(
670                    "ambiguous run input `{trimmed}` looks like a DSL key token; refusing to type it as literal text"
671                ),
672                None,
673                Some(trimmed),
674                Some("parse"),
675            )
676            .with_detail_entry("hint", json!(bare_key_token_hint(trimmed)))
677            .with_detail_entry("ambiguous_token", json!(trimmed))
678            .with_detail_entry("ambiguity", json!("bare_key_token")),
679        );
680    }
681
682    if let Some((verb, hint)) = reserved_verb_without_parens_hint(trimmed) {
683        return Some(
684            dsl_error(

695        );
696    }
697
698    None
699}

747fn reserved_verb_without_parens_hint(input: &str) -> Option<(&'static str, String)> {
748    let trimmed = input.trim();
749    let split_at = trimmed
750        .char_indices()
751        .find_map(|(index, character)| character.is_whitespace().then_some(index))?;
752    let verb = &trimmed[..split_at];
753    let rest = trimmed[split_at..].trim_start();
754    if !rest.starts_with('"') {
755        return None;
756    }
757
758    let canonical_verb = match verb.to_ascii_lowercase().as_str() {
759        "type" | "send" => "send",
760        "key" => "key",
761        "click" => "click",
762        "scroll" => "scroll",
763        _ => return None,
764    };
765
766    let quoted_hint = parse_quoted_string(rest, 0, input).ok().map_or_else(
767        || String::from("\"...\""),
768        |text| serde_json::to_string(&text).unwrap_or_else(|_| String::from("\"...\"")),
769    );
770    let hint = if canonical_verb == "send" {
771        format!(
772            "Use Tendril's canonical text action as `send({quoted_hint})`; for example `send({quoted_hint}),Return` when text should be followed by Return."
773        )
774    } else {
775        format!(
776            "Tendril DSL actions require parentheses. Use `send({quoted_hint})` for text, or rewrite the `{canonical_verb}` action with explicit Tendril DSL syntax."
777        )
778    };
779
780    Some((canonical_verb, hint))
781}

783fn split_top_level(input: &str) -> Result<Vec<&str>, TendrilError> {
784    let mut parts = Vec::new();
785    let mut start = 0_usize;
786    let mut in_string = false;
787    let mut escape = false;
788    let mut depth = 0_i32;
789
790    for (index, character) in input.char_indices() {
791        if in_string {
792            if escape {
793                escape = false;
794                continue;
795            }
796            match character {
797                '\\' => escape = true,
798                '"' => in_string = false,
799                _ => {}
800            }
801            continue;
802        }
803
804        match character {
805            '"' => in_string = true,
806            '(' => depth += 1,
807            ')' => {
808                if depth == 0 {
809                    return Err(dsl_error(
810                        "unexpected `)` in input sequence",
811                        None,
812                        None,
813                        Some("parse"),
814                    )
815                    .with_detail_entry("offset", json!(index)));
816                }
817                depth -= 1;
818            }
819            ';' if depth == 0 => {
820                return Err(dsl_error(
821                    "unexpected `;`; the DSL separator is `,`",
822                    Some(parts.len()),
823                    None,
824                    Some("parse"),
825                )
826                .with_detail_entry("offset", json!(index)));
827            }
828            ',' if depth == 0 => {
829                let part = input[start..index].trim();
830                if part.is_empty() {
831                    return Err(dsl_error(
832                        "empty action in DSL sequence",
833                        Some(parts.len()),
834                        None,
835                        Some("parse"),
836                    )
837                    .with_detail_entry("offset", json!(index)));
838                }
839                parts.push(part);
840                start = index + character.len_utf8();
841            }
842            _ => {}
843        }
844    }
845
846    if in_string {
847        return Err(dsl_error(
848            "unterminated string literal in input sequence",
849            None,
850            None,
851            Some("parse"),
852        ));
853    }
854
855    if depth != 0 {
856        return Err(dsl_error(
857            "unbalanced parentheses in input sequence",
858            None,
859            None,
860            Some("parse"),
861        ));
862    }
863
864    let tail = input[start..].trim();
865    if tail.is_empty() {
866        return Err(dsl_error(
867            "empty action in DSL sequence",
868            Some(parts.len()),
869            None,
870            Some("parse"),
871        ));
872    }
873    parts.push(tail);
874
875    Ok(parts)
876}

878#[allow(clippy::too_many_lines)]
879fn parse_action_segment(segment: &str, action_index: usize) -> Result<InputAction, TendrilError> {
880    if !segment.contains('(') {
881        let key = parse_key_token(segment).ok_or_else(|| {
882            dsl_error(
883                format!("invalid key tap `{segment}`"),
884                Some(action_index),
885                Some(segment),
886                Some("parse"),
887            )
888        })?;
889        return Ok(InputAction::KeyTap { key });
890    }
891
892    let open = segment.find('(').ok_or_else(|| {
893        dsl_error(
894            format!("invalid DSL action `{segment}`"),
895            Some(action_index),
896            Some(segment),
897            Some("parse"),
898        )
899    })?;
900    let close = segment.rfind(')').ok_or_else(|| {
901        dsl_error(
902            format!("missing closing `)` in `{segment}`"),
903            Some(action_index),
904            Some(segment),
905            Some("parse"),
906        )
907    })?;
908
909    if !segment[close + 1..].trim().is_empty() {
910        return Err(dsl_error(
911            format!("unexpected trailing input after `{segment}`"),
912            Some(action_index),
913            Some(segment),
914            Some("parse"),
915        ));
916    }
917
918    let name = segment[..open].trim().to_ascii_lowercase();
919    let inner = &segment[open + 1..close];
920
921    match name.as_str() {
922        "hold" => {
923            let argument = expect_single_argument(inner, action_index, segment)?;
924            Ok(InputAction::Hold {
925                modifier: parse_modifier(argument, action_index, segment)?,
926            })
927        }
928        "release" => {
929            let argument = expect_single_argument(inner, action_index, segment)?;
930            Ok(InputAction::Release {
931                modifier: parse_modifier(argument, action_index, segment)?,
932            })
933        }
934        "send" => {
935            let argument = expect_single_argument(inner, action_index, segment)?;
936            let text = parse_quoted_string(argument, action_index, segment)?;
937            if text.is_empty() {
938                return Err(dsl_error(
939                    "send(\"...\") requires a non-empty string",
940                    Some(action_index),
941                    Some(segment),
942                    Some("validate"),
943                ));
944            }
945            Ok(InputAction::Send { text })
946        }
947        "wait" => {
948            let argument = expect_single_argument(inner, action_index, segment)?;
949            Ok(InputAction::Wait {
950                duration_ms: parse_duration_ms(argument, action_index, segment)?,
951            })
952        }
953        "click" | "press" | "element" => {
954            let argument = expect_single_argument(inner, action_index, segment)?;
955            let id = parse_element_id(argument, action_index, segment)?;
956            Ok(InputAction::ElementClick { id })
957        }
958        "launch" | "open" | "switch" => {
959            let argument = expect_single_argument(inner, action_index, segment)?;
960            let package = parse_element_id(argument, action_index, segment)?;
961            Ok(InputAction::ElementClick {
962                id: format!("{name}:{package}"),
963            })
964        }
965        "tap_text" | "taptext" => {
966            let argument = expect_single_argument(inner, action_index, segment)?;
967            let text = parse_element_id(argument, action_index, segment)?;
968            Ok(InputAction::ElementClick {
969                id: format!("text={text}"),
970            })
971        }
972        "tap_desc" | "tap_description" | "tap_content_desc" => {
973            let argument = expect_single_argument(inner, action_index, segment)?;
974            let text = parse_element_id(argument, action_index, segment)?;
975            Ok(InputAction::ElementClick {
976                id: format!("desc={text}"),
977            })
978        }
979        "tap_resource" | "tap_resource_id" | "tap_id" => {
980            let argument = expect_single_argument(inner, action_index, segment)?;
981            let text = parse_element_id(argument, action_index, segment)?;
982            Ok(InputAction::ElementClick {
983                id: format!("resource={text}"),
984            })
985        }
986        "scroll_until" | "scrolluntil" => {
987            let argument = expect_single_argument(inner, action_index, segment)?;
988            let selector = parse_element_id(argument, action_index, segment)?;
989            Ok(InputAction::ElementClick {
990                id: format!("scroll-until:{selector}"),
991            })
992        }
993        "assert_visible" | "assertvisible" => {
994            let argument = expect_single_argument(inner, action_index, segment)?;
995            let selector = parse_element_id(argument, action_index, segment)?;
996            Ok(InputAction::ElementClick {
997                id: format!("assert-visible:{selector}"),
998            })
999        }
1000        "assert_absent" | "assertabsent" => {
1001            let argument = expect_single_argument(inner, action_index, segment)?;
1002            let selector = parse_element_id(argument, action_index, segment)?;
1003            Ok(InputAction::ElementClick {
1004                id: format!("assert-absent:{selector}"),
1005            })
1006        }
1007        "back" | "home" | "recents" | "assistant" | "notifications" | "quicksettings"
1008        | "status" => {
1009            if !inner.trim().is_empty() {
1010                return Err(dsl_error(
1011                    format!("`{name}` expects no arguments"),
1012                    Some(action_index),
1013                    Some(segment),
1014                    Some("parse"),
1015                ));
1016            }
1017            Ok(InputAction::ElementClick {
1018                id: format!("android:{name}"),
1019            })
1020        }
1021        "lclick" | "rclick" | "mclick" => {
1022            let arguments = split_arguments(inner, action_index, segment)?;
1023            if arguments.len() != 2 {
1024                return Err(dsl_error(
1025                    format!("`{name}` expects exactly two coordinates"),
1026                    Some(action_index),
1027                    Some(segment),
1028                    Some("parse"),
1029                ));
1030            }
1031            Ok(InputAction::Click {
1032                button: match name.as_str() {
1033                    "lclick" => MouseButton::Left,
1034                    "rclick" => MouseButton::Right,
1035                    _ => MouseButton::Middle,
1036                },
1037                x: parse_i32(arguments[0], action_index, segment, "x")?,
1038                y: parse_i32(arguments[1], action_index, segment, "y")?,
1039            })
1040        }
1041        "move" | "hover" => {
1042            let arguments = split_arguments(inner, action_index, segment)?;
1043            if arguments.len() != 2 {
1044                return Err(dsl_error(
1045                    format!("`{name}` expects exactly two coordinates"),
1046                    Some(action_index),
1047                    Some(segment),
1048                    Some("parse"),
1049                ));
1050            }
1051            Ok(InputAction::PointerMove {
1052                x: parse_i32(arguments[0], action_index, segment, "x")?,
1053                y: parse_i32(arguments[1], action_index, segment, "y")?,
1054            })
1055        }
1056        "dblclick" | "doubleclick" => {
1057            let arguments = split_arguments(inner, action_index, segment)?;
1058            if arguments.len() != 2 {
1059                return Err(dsl_error(
1060                    format!("`{name}` expects exactly two coordinates"),
1061                    Some(action_index),
1062                    Some(segment),
1063                    Some("parse"),
1064                ));
1065            }
1066            Ok(InputAction::DoubleClick {
1067                x: parse_i32(arguments[0], action_index, segment, "x")?,
1068                y: parse_i32(arguments[1], action_index, segment, "y")?,
1069            })
1070        }
1071        "drag" => {
1072            let arguments = split_arguments(inner, action_index, segment)?;
1073            if arguments.len() != 4 {
1074                return Err(dsl_error(
1075                    "`drag` expects exactly four coordinates",
1076                    Some(action_index),
1077                    Some(segment),
1078                    Some("parse"),
1079                ));
1080            }
1081            Ok(InputAction::Drag {
1082                x0: parse_i32(arguments[0], action_index, segment, "x0")?,
1083                y0: parse_i32(arguments[1], action_index, segment, "y0")?,
1084                x1: parse_i32(arguments[2], action_index, segment, "x1")?,
1085                y1: parse_i32(arguments[3], action_index, segment, "y1")?,
1086            })
1087        }
1088        "scroll" => {
1089            let arguments = split_arguments(inner, action_index, segment)?;
1090            if arguments.len() != 3 {
1091                return Err(dsl_error(
1092                    "`scroll` expects exactly x, y, and dy arguments",
1093                    Some(action_index),
1094                    Some(segment),
1095                    Some("parse"),
1096                ));
1097            }
1098            Ok(InputAction::Scroll {
1099                x: parse_i32(arguments[0], action_index, segment, "x")?,
1100                y: parse_i32(arguments[1], action_index, segment, "y")?,
1101                dy: parse_scroll_delta(arguments[2], action_index, segment)?,
1102            })
1103        }
1104        _ => Err(dsl_error(
1105            format!("unknown DSL action `{name}`"),
1106            Some(action_index),
1107            Some(segment),
1108            Some("parse"),
1109        )),
1110    }
1111}

1247fn parse_quoted_string(
1248    input: &str,
1249    action_index: usize,
1250    segment: &str,
1251) -> Result<String, TendrilError> {
1252    let trimmed = input.trim();
1253    if !(trimmed.starts_with('"') && trimmed.ends_with('"') && trimmed.len() >= 2) {
1254        return Err(dsl_error(
1255            "send(...) expects a double-quoted string literal",
1256            Some(action_index),
1257            Some(segment),
1258            Some("parse"),
1259        ));
1260    }
1261
1262    let mut result = String::new();
1263    let mut escape = false;
1264    for character in trimmed[1..trimmed.len() - 1].chars() {
1265        if escape {
1266            match character {
1267                '"' => result.push('"'),
1268                '\\' => result.push('\\'),
1269                'n' => result.push('\n'),
1270                'r' => result.push('\r'),
1271                't' => result.push('\t'),
1272                other => {
1273                    return Err(dsl_error(
1274                        format!("unsupported escape `\\{other}` in string literal"),
1275                        Some(action_index),
1276                        Some(segment),
1277                        Some("parse"),
1278                    ));
1279                }
1280            }
1281            escape = false;
1282            continue;
1283        }
1284
1285        if character == '\\' {
1286            escape = true;
1287        } else {
1288            result.push(character);
1289        }
1290    }
1291
1292    if escape {
1293        return Err(dsl_error(
1294            "unterminated escape sequence in string literal",
1295            Some(action_index),
1296            Some(segment),
1297            Some("parse"),
1298        ));
1299    }
1300
1301    Ok(result)
1302}

1304fn parse_duration_ms(input: &str, action_index: usize, segment: &str) -> Result<u64, TendrilError> {
1305    let trimmed = input.trim();
1306    if trimmed.is_empty() {
1307        return Err(dsl_error(
1308            "wait(...) requires a duration",
1309            Some(action_index),
1310            Some(segment),
1311            Some("parse"),
1312        ));
1313    }
1314
1315    let (number, unit) = if let Some(value) = trimmed.strip_suffix("ms") {
1316        (value.trim(), "ms")
1317    } else if let Some(value) = trimmed.strip_suffix('s') {
1318        (value.trim(), "s")
1319    } else {
1320        (trimmed, "ms")
1321    };
1322
1323    if number.is_empty() {
1324        return Err(dsl_error(
1325            "wait(...) requires a numeric duration",
1326            Some(action_index),
1327            Some(segment),
1328            Some("parse"),
1329        ));
1330    }
1331
1332    let millis = if unit == "ms" {
1333        number.parse::<u64>().map_err(|_| {
1334            dsl_error(
1335                format!("invalid millisecond duration `{trimmed}`"),
1336                Some(action_index),
1337                Some(segment),
1338                Some("parse"),
1339            )
1340        })?
1341    } else {
1342        let seconds = number.parse::<f64>().map_err(|_| {
1343            dsl_error(
1344                format!("invalid second duration `{trimmed}`"),
1345                Some(action_index),
1346                Some(segment),
1347                Some("parse"),
1348            )
1349        })?;
1350        if !seconds.is_finite() || seconds <= 0.0 {
1351            return Err(dsl_error(
1352                format!("duration `{trimmed}` must be greater than zero"),
1353                Some(action_index),
1354                Some(segment),
1355                Some("validate"),
1356            ));
1357        }
1358        u64::try_from(Duration::from_secs_f64(seconds).as_millis()).map_err(|_| {
1359            dsl_error(
1360                format!("duration `{trimmed}` is too large"),
1361                Some(action_index),
1362                Some(segment),
1363                Some("validate"),
1364            )
1365        })?
1366    };
1367
1368    if millis == 0 {
1369        return Err(dsl_error(
1370            format!("duration `{trimmed}` must be greater than zero"),
1371            Some(action_index),
1372            Some(segment),
1373            Some("validate"),
1374        ));
1375    }
1376
1377    Ok(millis)
1378}

1396fn parse_scroll_delta(
1397    input: &str,
1398    action_index: usize,
1399    segment: &str,
1400) -> Result<i32, TendrilError> {
1401    let dy = parse_i32(input, action_index, segment, "dy")?;
1402    if dy == 0 {
1403        return Err(dsl_error(
1404            "scroll dy must be non-zero",
1405            Some(action_index),
1406            Some(segment),
1407            Some("validate"),
1408        ));
1409    }
1410    if dy.unsigned_abs() > MAX_SCROLL_TICKS {
1411        return Err(dsl_error(
1412            format!(
1413                "scroll dy must be between -{MAX_SCROLL_TICKS} and {MAX_SCROLL_TICKS} wheel ticks"
1414            ),
1415            Some(action_index),
1416            Some(segment),
1417            Some("validate"),
1418        ));
1419    }
1420    Ok(dy)
1421}

1423fn dsl_error(
1424    message: impl Into<String>,
1425    action_index: Option<usize>,
1426    action: Option<&str>,
1427    stage: Option<&str>,
1428) -> TendrilError {
1429    let mut error = TendrilError::validation(message)
1430        .with_code("invalid_run_input")
1431        .with_field("input_definition");
1432    if let Some(action_index) = action_index {
1433        error = error
1434            .with_detail_entry("action_index", json!(action_index))
1435            .with_detail_entry("action_number", json!(action_index + 1));
1436    }
1437    if let Some(action) = action {
1438        error = error.with_detail_entry("action", json!(action));
1439    }
1440    if let Some(stage) = stage {
1441        error = error.with_detail_entry("stage", json!(stage));
1442    }
1443    error
1444}

1447fn scaled_coordinate(value: i32, numerator: u32, denominator: u32) -> i32 {
1448    if denominator == 0 {
1449        return value;
1450    }
1451
1452    let scaled = (i64::from(value) * i64::from(numerator) + i64::from(denominator / 2))
1453        / i64::from(denominator);
1454    i32::try_from(scaled).unwrap_or(if scaled.is_negative() {
1455        i32::MIN
1456    } else {
1457        i32::MAX
1458    })
1459}
