37#[serde(rename_all = "snake_case")]
38pub enum ClipboardSelection {
39    Clipboard,
40    Primary,
41}
42
43impl ClipboardSelection {
44    pub fn parse(value: Option<&str>) -> Result<Self, TendrilError> {
45        match value
46            .unwrap_or("clipboard")
47            .trim()
48            .to_ascii_lowercase()
49            .as_str()
50        {
51            "clipboard" => Ok(Self::Clipboard),
52            "primary" => Ok(Self::Primary),
53            other => Err(TendrilError::validation(format!(
54                "unsupported clipboard selection `{other}`; expected `clipboard` or `primary`"
55            ))
56            .with_code("invalid_clipboard_input")
57            .with_field("selection")),
58        }
59    }
60
61    fn as_str(self) -> &'static str {
62        match self {
63            Self::Clipboard => "clipboard",
64            Self::Primary => "primary",
65        }
66    }
67}

209fn ensure_x11_clipboard_supported(context: &AdapterContext) -> Result<(), TendrilError> {
210    if context.platform == PlatformKind::Linux && context.session == DesktopSession::X11 {
211        return Ok(());
212    }
213
214    Err(TendrilError::unsupported_capability(
215        "clipboard_not_supported",
216        "explicit clipboard get/set is currently implemented for Linux/X11 selections only",
217        Some(json!({
218            "platform": context.platform,
219            "session": context.session,
220            "supported": [{"platform": "linux", "session": "x11"}],
221            "suggested_action": "Use the headless X11 micro-environment for deterministic browser↔OS clipboard smokes. For Wayland/macOS/Windows, use the platform clipboard tool directly or file a platform-specific Tendril clipboard backend bead."
222        })),
223    ))
224}

302    impl X11ClipboardConnection {
303        fn connect() -> Result<Self, TendrilError> {
304            let (conn, screen_num) = x11rb::connect(None).map_err(|error| {
305                TendrilError::execution_failure(
306                    "clipboard_x11_connect_failed",
307                    format!(
308                        "failed to connect to the active X11 display for clipboard access: {error}"
309                    ),
310                    None,
311                )
312            })?;
313            let atoms = Atoms::new(&conn)
314                .map_err(|error| {
315                    TendrilError::execution_failure(
316                        "clipboard_x11_atom_failed",
317                        format!("failed to request X11 clipboard atoms: {error}"),
318                        None,
319                    )
320                })?
321                .reply()
322                .map_err(|error| {
323                    TendrilError::execution_failure(
324                        "clipboard_x11_atom_failed",
325                        format!("failed to intern X11 clipboard atoms: {error}"),
326                        None,
327                    )
328                })?;
329            Ok(Self {
330                conn,
331                screen_num,
332                atoms,
333            })
334        }
335
336        fn root(&self) -> Result<Window, TendrilError> {
337            self.conn
338                .setup()
339                .roots
340                .get(self.screen_num)
341                .map(|screen| screen.root)
342                .ok_or_else(|| {
343                    TendrilError::execution_failure(
344                        "clipboard_x11_screen_missing",
345                        format!(
346                            "X11 screen index `{}` was not present in setup",
347                            self.screen_num
348                        ),
349                        None,
350                    )
351                })
352        }
353
354        fn create_helper_window(&self) -> Result<Window, TendrilError> {
355            let window = self.conn.generate_id().map_err(|error| {
356                TendrilError::execution_failure(
357                    "clipboard_x11_window_failed",
358                    format!("failed to allocate an X11 helper window id: {error}"),
359                    None,
360                )
361            })?;
362            self.conn
363                .create_window(
364                    0,
365                    window,
366                    self.root()?,
367                    0,
368                    0,
369                    1,
370                    1,
371                    0,
372                    WindowClass::COPY_FROM_PARENT,
373                    0,
374                    &CreateWindowAux::new().event_mask(EventMask::PROPERTY_CHANGE),
375                )
376                .map_err(|error| {
377                    TendrilError::execution_failure(
378                        "clipboard_x11_window_failed",
379                        format!("failed to create an X11 clipboard helper window: {error}"),
380                        None,
381                    )
382                })?;
383            Ok(window)
384        }

394    pub fn get_selection(
395        selection: ClipboardSelection,
396        timeout: Duration,
397    ) -> Result<(String, Vec<String>), TendrilError> {
398        let x11 = X11ClipboardConnection::connect()?;
399        let selection_atom = x11.selection_atom(selection);
400        let owner = x11
401            .conn
402            .get_selection_owner(selection_atom)
403            .map_err(|error| clipboard_error("clipboard_query_failed", selection, error))?
404            .reply()
405            .map_err(|error| clipboard_error("clipboard_query_failed", selection, error))?
406            .owner;
407        if owner == 0 {
408            return Err(TendrilError::execution_failure(
409                "clipboard_selection_unowned",
410                format!(
411                    "X11 {} selection has no owner; copy text in the source application first or use `tendril clipboard set --text ...`",
412                    selection.as_str()
413                ),
414                None,
415            )
416            .with_detail_entry("selection", json!(selection.as_str()))
417            .with_detail_entry(
418                "suggested_action",
419                json!("Copy from the browser with Ctrl+C, then immediately run `tendril clipboard get --json`; X11 selection ownership is process-owned and may disappear when the source exits. If a Firefox page observed a copy event but this error persists after a drag gesture, verify the text is actually selected (for a textarea, selectionStart must differ from selectionEnd and a highlight should be visible); Firefox can fire a copy event for an empty textarea selection without taking clipboard ownership. Use a slower text-baseline drag, drag from the end of the text back to the start, or focus the field and use Ctrl+A then Ctrl+C."),
420            ));
421        }
422
423        let window = x11.create_helper_window()?;
424        let property = x11.atoms.TENDRIL_CLIPBOARD;
425        let deadline = Instant::now() + timeout;
426        let mut attempted_targets = Vec::new();
427        for (target_atom, target_name) in [
428            (x11.atoms.UTF8_STRING, "UTF8_STRING"),
429            (AtomEnum::STRING.into(), "STRING"),
430            (x11.atoms.TEXT, "TEXT"),
431        ] {
432            attempted_targets.push(target_name.to_owned());
433            match read_target(
434                &x11,
435                window,
436                selection,
437                selection_atom,
438                target_atom,
439                property,
440                deadline,
441            ) {
442                Ok(Some(text)) => {
443                    let notes = vec![format!(
444                        "Read X11 {} selection from owner 0x{owner:x} using target {target_name}.",
445                        selection.as_str()
446                    )];
447                    let _ = x11.conn.destroy_window(window);
448                    let _ = x11.conn.flush();
449                    return Ok((text, notes));
450                }
451                Ok(None) => {}
452                Err(error) => {
453                    let _ = x11.conn.destroy_window(window);
454                    let _ = x11.conn.flush();
455                    return Err(error);
456                }
457            }
458        }
459
460        let _ = x11.conn.destroy_window(window);
461        let _ = x11.conn.flush();
462        Err(TendrilError::execution_failure(
463            "clipboard_conversion_failed",
464            format!(
465                "X11 {} selection owner 0x{owner:x} did not provide UTF-8 or STRING text data",
466                selection.as_str()
467            ),
468            None,
469        )
470        .with_detail_entry("selection", json!(selection.as_str()))
471        .with_detail_entry("owner", json!(format!("0x{owner:x}")))
472        .with_detail_entry("attempted_targets", json!(attempted_targets))
473        .with_detail_entry(
474            "suggested_action",
475            json!("The source application may own a non-text selection or use a large INCR transfer not yet supported by Tendril; retry with a plain text selection or file a backend-specific clipboard bead."),
476        ))
477    }

479    fn read_target(
480        x11: &X11ClipboardConnection,
481        window: Window,
482        selection_kind: ClipboardSelection,
483        selection: Atom,
484        target: Atom,
485        property: Atom,
486        deadline: Instant,
487    ) -> Result<Option<String>, TendrilError> {
488        x11.conn
489            .delete_property(window, property)
490            .map_err(|error| {
491                clipboard_error("clipboard_property_delete_failed", selection_kind, error)
492            })?;
493        x11.conn
494            .convert_selection(window, selection, target, property, CURRENT_TIME)
495            .map_err(|error| clipboard_error("clipboard_convert_failed", selection_kind, error))?;
496        x11.conn.flush().map_err(|error| {
497            TendrilError::execution_failure(
498                "clipboard_x11_flush_failed",
499                format!("failed to flush X11 clipboard conversion request: {error}"),
500                None,
501            )
502        })?;
503
504        let notify = wait_for_selection_notify(&x11.conn, window, selection, target, deadline)?;
505        if notify.property == 0 {
506            return Ok(None);
507        }
508
509        let reply = x11
510            .conn
511            .get_property(true, window, notify.property, AtomEnum::ANY, 0, u32::MAX)
512            .map_err(|error| {
513                clipboard_error("clipboard_property_read_failed", selection_kind, error)
514            })?
515            .reply()
516            .map_err(|error| {
517                clipboard_error("clipboard_property_read_failed", selection_kind, error)
518            })?;
519
520        if reply.type_ == x11.atoms.TARGETS {
521            return Ok(None);
522        }
523        if atom_name(&x11.conn, reply.type_).as_deref() == Some("INCR") {
524            return Err(TendrilError::execution_failure(
525                "clipboard_incr_not_supported",
526                "X11 clipboard owner requested an incremental INCR transfer; Tendril currently supports direct text selections only",
527                None,
528            )
529            .with_detail_entry(
530                "suggested_action",
531                json!("Retry with a smaller plain-text clipboard payload or file a follow-up for X11 INCR clipboard transfers."),
532            ));
533        }
534
535        Ok(Some(String::from_utf8_lossy(&reply.value).to_string()))
536    }

538    fn wait_for_selection_notify(
539        conn: &x11rb::rust_connection::RustConnection,
540        window: Window,
541        selection: Atom,
542        target: Atom,
543        deadline: Instant,
544    ) -> Result<SelectionNotifyEvent, TendrilError> {
545        loop {
546            if Instant::now() >= deadline {
547                return Err(TendrilError::timeout(
548                    "clipboard_timeout",
549                    "timed out waiting for the X11 selection owner to provide clipboard data",
550                    Some(json!({
551                        "target": atom_name(conn, target),
552                        "suggested_action": "Verify the source application still owns the clipboard and retry with a plain text selection."
553                    })),
554                ));
555            }
556            if let Some(event) = conn.poll_for_event().map_err(|error| {
557                TendrilError::execution_failure(
558                    "clipboard_x11_event_failed",
559                    format!("failed while polling X11 clipboard events: {error}"),
560                    None,
561                )
562            })? {
563                if let Event::SelectionNotify(event) = event
564                    && event.requestor == window
565                    && event.selection == selection
566                    && event.target == target
567                {
568                    return Ok(event);
569                }
570            } else {
571                std::thread::sleep(CLIPBOARD_POLL_INTERVAL);
572            }
573        }
574    }

585    pub fn set_selection_in_background(
586        selection: ClipboardSelection,
587        text: &str,
588        serve_for: Duration,
589    ) -> Result<super::X11ClipboardServeHandle, TendrilError> {
590        let (ready_tx, ready_rx) = mpsc::channel();
591        let text = text.to_owned();
592        let join = thread::spawn(move || {
593            let owned = match own_selection(selection, serve_for) {
594                Ok(owned) => owned,
595                Err(error) => {
596                    let _ = ready_tx.send(Err(error));
597                    return Ok((0, Vec::new()));
598                }
599            };
600            if ready_tx.send(Ok(())).is_err() {
601                return serve_owned_selection(owned, &text, serve_for);
602            }
603            serve_owned_selection(owned, &text, serve_for)
604        });
605
606        match ready_rx.recv() {
607            Ok(Ok(())) => Ok(super::X11ClipboardServeHandle { join }),
608            Ok(Err(error)) => {
609                let _ = join.join();
610                Err(error)
611            }
612            Err(error) => {
613                let _ = join.join();
614                Err(TendrilError::execution_failure(
615                    "clipboard_serve_failed",
616                    format!(
617                        "X11 clipboard owner thread exited before reporting readiness: {error}"
618                    ),
619                    None,
620                ))
621            }
622        }
623    }

633    fn own_selection(
634        selection: ClipboardSelection,
635        serve_for: Duration,
636    ) -> Result<OwnedSelection, TendrilError> {
637        let x11 = X11ClipboardConnection::connect()?;
638        let selection_atom = x11.selection_atom(selection);
639        let window = x11.create_helper_window()?;
640        x11.conn
641            .set_selection_owner(window, selection_atom, CURRENT_TIME)
642            .map_err(|error| clipboard_error("clipboard_set_owner_failed", selection, error))?;
643        x11.conn.flush().map_err(|error| {
644            TendrilError::execution_failure(
645                "clipboard_x11_flush_failed",
646                format!("failed to flush X11 clipboard ownership request: {error}"),
647                None,
648            )
649        })?;
650        let owner = x11
651            .conn
652            .get_selection_owner(selection_atom)
653            .map_err(|error| clipboard_error("clipboard_query_failed", selection, error))?
654            .reply()
655            .map_err(|error| clipboard_error("clipboard_query_failed", selection, error))?
656            .owner;
657        if owner != window {
658            return Err(TendrilError::execution_failure(
659                "clipboard_set_owner_failed",
660                format!(
661                    "failed to own X11 {} selection; current owner is 0x{owner:x}, expected helper window 0x{window:x}",
662                    selection.as_str()
663                ),
664                None,
665            )
666            .with_detail_entry("selection", json!(selection.as_str())));
667        }
668
669        let mut notes = vec![format!(
670            "Owned X11 {} selection as helper window 0x{window:x}; serving requests for {}ms.",
671            selection.as_str(),
672            serve_for.as_millis()
673        )];
674        if serve_for.is_zero() {
675            notes.push(
676                "serve_ms=0 releases the selection when this process exits; consumers must request before exit for a persistent paste.".to_owned(),
677            );
678        }
679
680        Ok(OwnedSelection {
681            x11,
682            selection,
683            selection_atom,
684            window,
685            notes,
686        })
687    }

689    fn serve_owned_selection(
690        owned: OwnedSelection,
691        text: &str,
692        serve_for: Duration,
693    ) -> Result<(usize, Vec<String>), TendrilError> {
694        let OwnedSelection {
695            x11,
696            selection,
697            selection_atom,
698            window,
699            mut notes,
700        } = owned;
701        let deadline = Instant::now() + serve_for;
702        let mut served_requests = 0;
703        while Instant::now() < deadline {
704            if let Some(event) = x11.conn.poll_for_event().map_err(|error| {
705                TendrilError::execution_failure(
706                    "clipboard_x11_event_failed",
707                    format!("failed while polling X11 clipboard events: {error}"),
708                    None,
709                )
710            })? {
711                match event {
712                    Event::SelectionRequest(request)
713                        if request.owner == window && request.selection == selection_atom =>
714                    {
715                        serve_selection_request(&x11, selection, &request, text)?;
716                        served_requests += 1;
717                    }
718                    Event::SelectionClear(event) if event.owner == window => {
719                        notes.push(
720                            "Selection ownership was replaced by another X11 client before serve_ms elapsed.".to_owned(),
721                        );
722                        break;
723                    }
724                    _ => {}
725                }
726            } else {
727                std::thread::sleep(CLIPBOARD_POLL_INTERVAL);
728            }
729        }
730
731        let _ = x11.conn.destroy_window(window);
732        let _ = x11.conn.flush();
733        Ok((served_requests, notes))
734    }
735
736    fn serve_selection_request(
737        x11: &X11ClipboardConnection,
738        selection: ClipboardSelection,
739        request: &SelectionRequestEvent,
740        text: &str,
741    ) -> Result<(), TendrilError> {
742        let property = if request.property == 0 {
743            request.target
744        } else {
745            request.property
746        };
747        let mut response_property = property;
748
749        if request.target == x11.atoms.TARGETS {
750            let targets = [
751                x11.atoms.UTF8_STRING,
752                AtomEnum::STRING.into(),
753                x11.atoms.TEXT,
754                x11.atoms.TARGETS,
755            ];
756            x11.conn
757                .change_property32(
758                    PropMode::REPLACE,
759                    request.requestor,
760                    property,
761                    AtomEnum::ATOM,
762                    &targets,
763                )
764                .map_err(|error| clipboard_error("clipboard_serve_failed", selection, error))?;
765        } else if request.target == x11.atoms.UTF8_STRING || request.target == x11.atoms.TEXT {
766            x11.conn
767                .change_property8(
768                    PropMode::REPLACE,
769                    request.requestor,
770                    property,
771                    x11.atoms.UTF8_STRING,
772                    text.as_bytes(),
773                )
774                .map_err(|error| clipboard_error("clipboard_serve_failed", selection, error))?;
775        } else if request.target == u32::from(AtomEnum::STRING) {
776            x11.conn
777                .change_property8(
778                    PropMode::REPLACE,
779                    request.requestor,
780                    property,
781                    AtomEnum::STRING,
782                    text.as_bytes(),
783                )
784                .map_err(|error| clipboard_error("clipboard_serve_failed", selection, error))?;
785        } else {
786            response_property = AtomEnum::NONE.into();
787        }
788
789        let notify = SelectionNotifyEvent {
790            response_type: x11rb::protocol::xproto::SELECTION_NOTIFY_EVENT,
791            sequence: 0,
792            time: request.time,
793            requestor: request.requestor,
794            selection: request.selection,
795            target: request.target,
796            property: response_property,
797        };
798        x11.conn
799            .send_event(false, request.requestor, EventMask::NO_EVENT, notify)
800            .map_err(|error| clipboard_error("clipboard_serve_notify_failed", selection, error))?;
801        x11.conn.flush().map_err(|error| {
802            TendrilError::execution_failure(
803                "clipboard_x11_flush_failed",
804                format!("failed to flush X11 clipboard response: {error}"),
805                None,
806            )
807        })?;
808        Ok(())
809    }
810
811    fn clipboard_error(
812        code: &'static str,
813        selection: ClipboardSelection,
814        error: impl std::fmt::Display,
815    ) -> TendrilError {
816        TendrilError::execution_failure(
817            code,
818            format!(
819                "X11 {} clipboard operation failed: {error}",
820                selection.as_str()
821            ),
822            None,
823        )
824        .with_detail_entry("selection", json!(selection.as_str()))
825    }
826
827    fn atom_name(conn: &x11rb::rust_connection::RustConnection, atom: Atom) -> Option<String> {
828        if atom == 0 {
829            return None;
830        }
831        conn.get_atom_name(atom)
832            .ok()?
833            .reply()
834            .ok()
835            .and_then(|reply| String::from_utf8(reply.name).ok())
836            .filter(|name| !name.is_empty())
837    }
