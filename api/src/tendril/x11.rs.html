97pub(crate) fn execute_input(
98    platform: PlatformKind,
99    x11_display: Option<&str>,
100    request: &InputRequest,
101) -> Result<InputOutcome, TendrilError> {
102    let context = AdapterContext::linux(crate::platform::DesktopSession::X11, None)
103        .with_x11_display(x11_display.map(str::to_owned));
104    let connection = X11Connection::connect(&context, AdapterOperation::InputControl)
105        .map_err(TendrilError::from)?;
106
107    ensure_xtest_available(&connection, platform)?;
108
109    let mut run_state = prepare_x11_input_state(&connection, request)?;
110    let keyboard_map = KeyboardMap::load(&connection).map_err(|error| {
111        input_execution_error(
112            "keyboard_mapping_failed",
113            format!("failed to load the X11 keyboard mapping: {error}"),
114            None,
115            None,
116        )
117    })?;
118    let mut held_modifiers = HashSet::new();
119
120    if let Some(text) = &request.text {
121        let dispatch_notes = type_text(
122            &connection,
123            &keyboard_map,
124            request,
125            text,
126            &held_modifiers,
127            Some(0),
128            Some("text"),
129        )?;
130        run_state.notes.extend(dispatch_notes);
131        flush_x11_input(&connection, Some(0), Some("text"), "text events")?;
132        return Ok(run_state.finish(&connection, request, 1));
133    }
134
135    let dispatch_notes =
136        dispatch_x11_actions(&connection, &keyboard_map, request, &mut held_modifiers)?;
137    run_state.notes.extend(dispatch_notes);
138    Ok(run_state.finish(&connection, request, request.actions.len()))
139}

230fn prepare_x11_input_state(
231    connection: &X11Connection,
232    request: &InputRequest,
233) -> Result<X11InputRunState, TendrilError> {
234    let mut notes = Vec::new();
235    let mut restore_error = None;
236    let restore_state = capture_requested_restore_state(
237        connection,
238        request.restore_focus,
239        &mut notes,
240        &mut restore_error,
241    );
242    let previous_focus = restore_state
243        .as_ref()
244        .map(|state| focus_snapshot_for_window(connection, state.window));
245
246    let keyboard_input = request.text.is_some() || request.actions.iter().any(action_is_keyboard);
247    let window_target = matches!(request.target, CaptureTargetKind::Window);
248    let focus_required = keyboard_input || window_target;
249    let focus_transferred = if window_target {
250        prepare_window_focus(connection, request, &mut notes)?
251    } else if keyboard_input {
252        notes.push(
253            "Display-scoped keyboard input uses the currently focused control; place focus explicitly if a different app should receive text or key taps."
254                .to_owned(),
255        );
256        false
257    } else {
258        false
259    };
260
261    Ok(X11InputRunState {
262        focus_required,
263        focus_transferred,
264        previous_focus,
265        restore_state,
266        restore_error,
267        notes,
268    })
269}

271fn capture_requested_restore_state(
272    connection: &X11Connection,
273    restore_focus: bool,
274    notes: &mut Vec<String>,
275    restore_error: &mut Option<String>,
276) -> Option<X11RestoreState> {
277    if !restore_focus {
278        notes.push(
279            "Focus restoration disabled for this run; focus may remain on the target.".to_owned(),
280        );
281        return None;
282    }
283
284    match capture_x11_restore_state(connection) {
285        Ok(Some(state)) => {
286            notes.push(format!(
287                "Captured previous X11 focus {} and pointer position for post-run restoration.",
288                format_window_id(state.window)
289            ));
290            Some(state)
291        }
292        Ok(None) => {
293            let message = "X11 did not report a restorable active window before input".to_owned();
294            notes.push(format!(
295                "Focus restoration requested but skipped: {message}."
296            ));
297            *restore_error = Some(message);
298            None
299        }
300        Err(error) => {
301            notes.push(format!(
302                "Focus restoration requested but pre-run snapshot failed: {error}."
303            ));
304            *restore_error = Some(error);
305            None
306        }
307    }
308}

310fn prepare_window_focus(
311    connection: &X11Connection,
312    request: &InputRequest,
313    notes: &mut Vec<String>,
314) -> Result<bool, TendrilError> {
315    let window = parse_window_id(&request.target_id)
316        .map_err(|message| input_execution_error("invalid_target", message, None, Some("focus")))?;
317    activate_window(connection, window).map_err(|error| {
318        input_execution_error(
319            "focus_failed",
320            format!("failed to focus target window before input delivery: {error}"),
321            None,
322            Some("focus"),
323        )
324    })?;
325    notes.push(
326        "Activated the target X11 window before input delivery so mouse gestures and keyboard events are delivered to the requested window instead of the previously focused or stacked target."
327            .to_owned(),
328    );
329    std::thread::sleep(reliability_delay());
330    Ok(true)
331}

398fn restore_x11_state_if_requested(
399    connection: &X11Connection,
400    restore_state: Option<&X11RestoreState>,
401) -> X11RestoreOutcome {
402    let Some(state) = restore_state else {
403        return X11RestoreOutcome::default();
404    };
405
406    let mut outcome = X11RestoreOutcome::default();
407    let mut errors = Vec::new();
408    match activate_window(connection, state.window) {
409        Ok(()) => outcome.focus_restored = true,
410        Err(error) => errors.push(format!("failed to restore X11 focus: {error}")),
411    }
412
413    if let Some((x, y)) = state.pointer_root {
414        match move_pointer(connection, x, y, None, Some("restore_pointer")) {
415            Ok(()) => {
416                if let Err(error) = connection.conn.flush() {
417                    errors.push(format!(
418                        "failed to flush restored X11 pointer position: {error}"
419                    ));
420                } else {
421                    outcome.pointer_restored = true;
422                }
423            }
424            Err(error) => errors.push(format!("failed to restore X11 pointer position: {error}")),
425        }
426    }
427
428    if !errors.is_empty() {
429        outcome.error = Some(errors.join("; "));
430    }
431    outcome
432}

435    let reply = connection
436        .conn
437        .get_property(
438            false,
439            connection.screen.root,

441            AtomEnum::WINDOW,
442            0,

445        .map_err(|error| format!("failed to query _NET_ACTIVE_WINDOW: {error}"))?
446        .reply()
447        .map_err(|error| format!("failed to read _NET_ACTIVE_WINDOW: {error}"))?;
448    Ok(reply

450        .and_then(|mut values| values.next())
451        .filter(|window| *window != 0))

477fn discover_displays(
478    context: &AdapterContext,
479    connection: &X11Connection,
480) -> Result<Vec<TargetDescriptor>, PlatformAdapterError> {
481    if connection
482        .conn
483        .extension_information(randr::X11_EXTENSION_NAME)
484        .map_err(|error| adapter_failure(context, AdapterOperation::TargetDiscovery, error))?
485        .is_none()
486    {
487        return Ok(vec![fallback_display(connection)]);
488    }
489
490    let monitors = connection
491        .conn
492        .randr_get_monitors(connection.screen.root, true)
493        .map_err(|error| adapter_failure(context, AdapterOperation::TargetDiscovery, error))?
494        .reply()
495        .map_err(|error| adapter_failure(context, AdapterOperation::TargetDiscovery, error))?;
496
497    let mut displays = Vec::new();
498    for (index, monitor) in monitors.monitors.into_iter().enumerate() {
499        if monitor.width == 0 || monitor.height == 0 {
500            continue;
501        }
502        let id = connection
503            .atom_name(monitor.name)
504            .unwrap_or_else(|| format!("{}", index + 1));
505        displays.push(TargetDescriptor {
506            id: id.clone(),
507            title: None,
508            kind: CaptureTargetKind::Display,
509            name: id,
510            bounds: Bounds {
511                x: i32::from(monitor.x),
512                y: i32::from(monitor.y),
513                width: u32::from(monitor.width),
514                height: u32::from(monitor.height),
515            },
516            scale_factor: ScaleFactor::identity(),
517            capture_supported: true,
518            input_supported: true,
519            app_name: None,
520            process_id: None,
521            diagnostics: Vec::new(),
522        });
523    }
524
525    if displays.is_empty() {
526        displays.push(fallback_display(connection));
527    }
528
529    Ok(displays)
530}

577fn discover_window(connection: &X11Connection, window: Window) -> Option<TargetDescriptor> {
578    let attributes = connection
579        .conn
580        .get_window_attributes(window)
581        .ok()?
582        .reply()
583        .ok()?;
584    if attributes.map_state != MapState::VIEWABLE {
585        return None;
586    }
587
588    let geometry = connection.conn.get_geometry(window).ok()?.reply().ok()?;
589    if geometry.width == 0 || geometry.height == 0 {
590        return None;
591    }
592
593    let translated = connection
594        .conn
595        .translate_coordinates(window, connection.screen.root, 0, 0)
596        .ok()?
597        .reply()
598        .ok()?;
599
600    let title = connection
601        .text_property(window, connection.atoms._NET_WM_NAME)
602        .or_else(|| connection.text_property(window, connection.atoms.WM_NAME))
603        .filter(|value| !value.is_empty());
604    let app_name = connection.class_name(window);
605    let process_id = connection.cardinal_property(window, connection.atoms._NET_WM_PID);
606    if crate::discovery::is_filtered_system_window(app_name.as_deref(), title.as_deref()) {
607        return None;
608    }
609    let name = app_name
610        .clone()
611        .or_else(|| title.clone())
612        .unwrap_or_else(|| format_window_id(window));
613
614    Some(TargetDescriptor {
615        id: format_window_id(window),
616        title,
617        kind: CaptureTargetKind::Window,
618        name,
619        bounds: Bounds {
620            x: i32::from(translated.dst_x),
621            y: i32::from(translated.dst_y),
622            width: u32::from(geometry.width),
623            height: u32::from(geometry.height),
624        },
625        scale_factor: ScaleFactor::identity(),
626        capture_supported: true,
627        input_supported: true,
628        app_name,
629        process_id,
630        diagnostics: Vec::new(),
631    })
632}

634fn capture_window(
635    context: &AdapterContext,
636    connection: &X11Connection,
637    target_id: &str,
638) -> Result<Vec<u8>, PlatformAdapterError> {
639    let window = parse_window_id(target_id).map_err(|message| {
640        PlatformAdapterError::adapter_failure(AdapterOperation::Capture, context.platform, message)
641    })?;
642    let geometry = connection
643        .conn
644        .get_geometry(window)
645        .map_err(|error| adapter_failure(context, AdapterOperation::Capture, error))?
646        .reply()
647        .map_err(|error| adapter_failure(context, AdapterOperation::Capture, error))?;
648    if geometry.width == 0 || geometry.height == 0 {
649        return Err(PlatformAdapterError::adapter_failure(
650            AdapterOperation::Capture,
651            context.platform,
652            format!("window `{target_id}` has zero-sized bounds"),
653        ));
654    }
655
656    let occluding_windows = occluding_windows_above_target(context, connection, window)?;
657    let direct_capture = capture_region(
658        context,
659        connection,
660        window,
661        0,
662        0,
663        geometry.width,
664        geometry.height,
665        Some(connection.screen.root_visual),
666    )?;
667
668    let direct_is_black = capture_png_is_solid_black(&direct_capture).map_err(|error| {
669        PlatformAdapterError::adapter_failure(
670            AdapterOperation::Capture,
671            context.platform,
672            format!(
673                "failed to validate X11 window capture for `{target_id}` before returning it: {error}"
674            ),
675        )
676    })?;
677    if direct_is_black {
678        return capture_window_via_root_crop_after_raise(
679            context,
680            connection,
681            target_id,
682            window,
683            &geometry,
684            &X11WindowCaptureFallbackReason::SolidBlackWindowDrawable,
685        );
686    }
687
688    if !occluding_windows.is_empty() {
689        return capture_window_via_root_crop_after_raise(
690            context,
691            connection,
692            target_id,
693            window,
694            &geometry,
695            &X11WindowCaptureFallbackReason::OverlappingWindows(occluding_windows),
696        );
697    }
698
699    Ok(direct_capture)
700}

707    geometry: &x11rb::protocol::xproto::GetGeometryReply,
708    reason: &X11WindowCaptureFallbackReason,
709) -> Result<Vec<u8>, PlatformAdapterError> {
710    let reason_summary = reason.summary();
711    let restore_state = capture_x11_restore_state(connection).map_err(|error| {
712        PlatformAdapterError::adapter_failure(
713            AdapterOperation::Capture,
714            context.platform,
715            format!(
716                "X11 window capture for `{target_id}` detected {reason_summary}. Tendril would need to temporarily raise the target and capture a root-window crop, but it could not snapshot the current focus for restoration: {error}"
717            ),
718        )
719    })?;
720
721    activate_window(connection, window).map_err(|error| {
722        PlatformAdapterError::adapter_failure(
723            AdapterOperation::Capture,
724            context.platform,
725            format!(
726                "X11 window capture for `{target_id}` detected {reason_summary}. Tendril could not raise it for an unoccluded root-window crop fallback: {error}"
727            ),
728        )
729    })?;
730    std::thread::sleep(reliability_delay());
731
732    let fallback_result = capture_raised_window_from_root_crop(
733        context,
734        connection,
735        target_id,
736        window,
737        geometry.width,
738        geometry.height,
739        &reason_summary,
740    );
741    let restore_outcome = restore_x11_state_if_requested(connection, restore_state.as_ref());
742
743    match (fallback_result, restore_outcome.error) {
744        (Ok(image_bytes), None) => Ok(image_bytes),
745        (Ok(_), Some(restore_error)) => Err(PlatformAdapterError::adapter_failure(
746            AdapterOperation::Capture,
747            context.platform,
748            format!(
749                "X11 window capture for `{target_id}` recovered from {reason_summary} by raising and root-cropping the target, but focus restoration failed afterwards: {restore_error}"
750            ),
751        )),
752        (Err(error), None) => Err(error),
753        (Err(error), Some(restore_error)) => Err(PlatformAdapterError::adapter_failure(
754            AdapterOperation::Capture,
755            context.platform,
756            format!(
757                "{error}; additionally, focus restoration failed after the fallback attempt: {restore_error}"
758            ),
759        )),
760    }
761}

763fn capture_raised_window_from_root_crop(
764    context: &AdapterContext,
765    connection: &X11Connection,
766    target_id: &str,
767    window: Window,
768    width: u16,
769    height: u16,
770    reason_summary: &str,
771) -> Result<Vec<u8>, PlatformAdapterError> {
772    let translated = connection
773        .conn
774        .translate_coordinates(window, connection.screen.root, 0, 0)
775        .map_err(|error| adapter_failure(context, AdapterOperation::Capture, error))?
776        .reply()
777        .map_err(|error| adapter_failure(context, AdapterOperation::Capture, error))?;
778
779    let image_bytes = capture_region(
780        context,
781        connection,
782        connection.screen.root,
783        translated.dst_x,
784        translated.dst_y,
785        width,
786        height,
787        Some(connection.screen.root_visual),
788    )?;
789
790    let fallback_is_black = capture_png_is_solid_black(&image_bytes).map_err(|error| {
791        PlatformAdapterError::adapter_failure(
792            AdapterOperation::Capture,
793            context.platform,
794            format!(
795                "failed to validate X11 root-crop fallback capture for `{target_id}` before returning it: {error}"
796            ),
797        )
798    })?;
799    if fallback_is_black {
800        return Err(PlatformAdapterError::adapter_failure(
801            AdapterOperation::Capture,
802            context.platform,
803            format!(
804                "X11 window capture for `{target_id}` detected {reason_summary}, then the raise-plus-root-crop fallback was also solid black. The target may be minimized, outside the visible display, still occluded by an override-redirect window, or otherwise unavailable to XGetImage; capture a visible display target or make the window visible before retrying."
805            ),
806        ));
807    }
808
809    Ok(image_bytes)
810}

812fn capture_png_is_solid_black(image_bytes: &[u8]) -> Result<bool, String> {
813    let decoded = image::load_from_memory(image_bytes)
814        .map_err(|error| format!("captured PNG could not be decoded: {error}"))?;
815    let rgba = decoded.to_rgba8();
816    if rgba.width() == 0 || rgba.height() == 0 {
817        return Ok(false);
818    }
819
820    Ok(rgba.pixels().all(|pixel| {
821        pixel[0] <= SOLID_BLACK_CHANNEL_THRESHOLD
822            && pixel[1] <= SOLID_BLACK_CHANNEL_THRESHOLD
823            && pixel[2] <= SOLID_BLACK_CHANNEL_THRESHOLD
824    }))
825}

827fn occluding_windows_above_target(
828    context: &AdapterContext,
829    connection: &X11Connection,
830    target: Window,
831) -> Result<Vec<X11OccludingWindow>, PlatformAdapterError> {
832    let Some(target_bounds) = viewable_window_bounds(connection, target).map_err(|error| {
833        PlatformAdapterError::adapter_failure(
834            AdapterOperation::Capture,
835            context.platform,
836            format!("failed to inspect X11 target bounds for occlusion detection: {error}"),
837        )
838    })?
839    else {
840        return Ok(Vec::new());
841    };
842
843    let stacking = stacking_windows(context, connection)?;
844    let Some(target_index) = stacking.iter().position(|window| *window == target) else {
845        return Ok(Vec::new());
846    };
847
848    let mut occluding_windows = Vec::new();
849    for candidate in stacking.iter().skip(target_index + 1).copied() {
850        if candidate == target {
851            continue;
852        }
853        let Some(bounds) = viewable_window_bounds(connection, candidate).map_err(|error| {
854            PlatformAdapterError::adapter_failure(
855                AdapterOperation::Capture,
856                context.platform,
857                format!(
858                    "failed to inspect X11 stacking candidate {} for occlusion detection: {error}",
859                    format_window_id(candidate)
860                ),
861            )
862        })?
863        else {
864            continue;
865        };
866        if windows_overlap(&target_bounds, &bounds) {
867            occluding_windows.push(X11OccludingWindow {
868                id: format_window_id(candidate),
869                name: window_display_name(connection, candidate),
870                bounds,
871            });
872        }
873    }
874
875    Ok(occluding_windows)
876}

878fn stacking_windows(
879    context: &AdapterContext,
880    connection: &X11Connection,
881) -> Result<Vec<Window>, PlatformAdapterError> {
882    let windows = connection.window_list(context)?;
883    if !windows.is_empty() {
884        return Ok(windows);
885    }
886
887    connection
888        .conn
889        .query_tree(connection.screen.root)
890        .map_err(|error| adapter_failure(context, AdapterOperation::Capture, error))?
891        .reply()
892        .map(|reply| reply.children)
893        .map_err(|error| adapter_failure(context, AdapterOperation::Capture, error))
894}

896fn viewable_window_bounds(
897    connection: &X11Connection,
898    window: Window,
899) -> Result<Option<X11WindowBounds>, String> {
900    let attributes = connection
901        .conn
902        .get_window_attributes(window)
903        .map_err(|error| format!("failed to query window attributes: {error}"))?
904        .reply()
905        .map_err(|error| format!("failed to read window attributes: {error}"))?;
906    if attributes.map_state != MapState::VIEWABLE {
907        return Ok(None);
908    }
909
910    let geometry = connection
911        .conn
912        .get_geometry(window)
913        .map_err(|error| format!("failed to query window geometry: {error}"))?
914        .reply()
915        .map_err(|error| format!("failed to read window geometry: {error}"))?;
916    if geometry.width == 0 || geometry.height == 0 {
917        return Ok(None);
918    }
919
920    let translated = connection
921        .conn
922        .translate_coordinates(window, connection.screen.root, 0, 0)
923        .map_err(|error| format!("failed to translate window coordinates: {error}"))?
924        .reply()
925        .map_err(|error| format!("failed to read translated window coordinates: {error}"))?;
926
927    Ok(Some(X11WindowBounds {
928        x: i32::from(translated.dst_x),
929        y: i32::from(translated.dst_y),
930        width: u32::from(geometry.width),
931        height: u32::from(geometry.height),
932    }))
933}

935fn windows_overlap(left: &X11WindowBounds, right: &X11WindowBounds) -> bool {
936    let left_right = left.x.saturating_add(u32_to_i32_saturating(left.width));
937    let left_bottom = left.y.saturating_add(u32_to_i32_saturating(left.height));
938    let right_right = right.x.saturating_add(u32_to_i32_saturating(right.width));
939    let right_bottom = right.y.saturating_add(u32_to_i32_saturating(right.height));
940
941    left.x < right_right && right.x < left_right && left.y < right_bottom && right.y < left_bottom
942}
943
944fn u32_to_i32_saturating(value: u32) -> i32 {
945    i32::try_from(value).unwrap_or(i32::MAX)
946}

975fn capture_display(
976    context: &AdapterContext,
977    connection: &X11Connection,
978    target_id: &str,
979) -> Result<Vec<u8>, PlatformAdapterError> {
980    let mut displays = discover_displays(context, connection)?;
981    // Sort displays the same way sort_inventory does so numeric index lookup
982    // matches the sequential IDs assigned by discovery.
983    displays.sort_by(|left, right| {
984        let left_key = (
985            left.bounds.y,
986            left.bounds.x,
987            left.name.to_ascii_lowercase(),
988            left.id.to_ascii_lowercase(),
989        );
990        let right_key = (
991            right.bounds.y,
992            right.bounds.x,
993            right.name.to_ascii_lowercase(),
994            right.id.to_ascii_lowercase(),
995        );
996        left_key.cmp(&right_key)
997    });
998    // target_id is a 1-based numeric index assigned by sort_inventory.
999    let index: usize = target_id.parse().map_err(|_| {
1000        PlatformAdapterError::adapter_failure(
1001            AdapterOperation::Capture,
1002            context.platform,
1003            format!("display `{target_id}` is not a valid numeric display index"),
1004        )
1005    })?;
1006    let display = displays.get(index.saturating_sub(1)).ok_or_else(|| {
1007        PlatformAdapterError::adapter_failure(
1008            AdapterOperation::Capture,
1009            context.platform,
1010            format!("display `{target_id}` was not found during capture"),
1011        )
1012    })?;
1013
1014    let x = clamp_i16(display.bounds.x).map_err(|message| {
1015        PlatformAdapterError::adapter_failure(AdapterOperation::Capture, context.platform, message)
1016    })?;
1017    let y = clamp_i16(display.bounds.y).map_err(|message| {
1018        PlatformAdapterError::adapter_failure(AdapterOperation::Capture, context.platform, message)
1019    })?;
1020    let width = clamp_u16(display.bounds.width).map_err(|message| {
1021        PlatformAdapterError::adapter_failure(AdapterOperation::Capture, context.platform, message)
1022    })?;
1023    let height = clamp_u16(display.bounds.height).map_err(|message| {
1024        PlatformAdapterError::adapter_failure(AdapterOperation::Capture, context.platform, message)
1025    })?;
1026
1027    capture_region(
1028        context,
1029        connection,
1030        connection.screen.root,
1031        x,
1032        y,
1033        width,
1034        height,
1035        Some(connection.screen.root_visual),
1036    )
1037}

1040fn capture_region(
1041    context: &AdapterContext,
1042    connection: &X11Connection,
1043    drawable: Window,
1044    x: i16,
1045    y: i16,
1046    width: u16,
1047    height: u16,
1048    fallback_visual: Option<Visualid>,
1049) -> Result<Vec<u8>, PlatformAdapterError> {

1051        .map_err(|error| adapter_failure(context, AdapterOperation::Capture, error))?;
1052    let visual = if visual == 0 {
1053        fallback_visual
1054    } else {
1055        Some(visual)
1056    };

1058        PlatformAdapterError::adapter_failure(AdapterOperation::Capture, context.platform, error)

1062fn encode_png(
1063    image: &Image<'_>,
1064    connection: &X11Connection,
1065    visual: Option<Visualid>,
1066) -> Result<Vec<u8>, String> {
1067    let visual_id = visual.unwrap_or(connection.screen.root_visual);
1068    let visual = lookup_visual(connection.conn.setup(), visual_id)
1069        .ok_or_else(|| format!("failed to resolve X11 visual 0x{visual_id:x} for capture"))?;
1070    if !matches!(
1071        visual.class,
1072        VisualClass::TRUE_COLOR | VisualClass::DIRECT_COLOR
1073    ) {
1074        return Err(format!(
1075            "capture requires a TrueColor/DirectColor visual, got {:?}",
1076            visual.class
1077        ));
1078    }
1079    let pixel_layout = PixelLayout::from_visual_type(visual)
1080        .map_err(|error| format!("failed to derive pixel layout from X11 visual: {error}"))?;
1081
1082    let mut rgba = ImageBuffer::new(u32::from(image.width()), u32::from(image.height()));
1083    for y in 0..image.height() {
1084        for x in 0..image.width() {
1085            let (red, green, blue) = pixel_layout.decode(image.get_pixel(x, y));
1086            rgba.put_pixel(
1087                u32::from(x),
1088                u32::from(y),
1089                Rgba([
1090                    u8::try_from(red >> 8).unwrap_or(0),
1091                    u8::try_from(green >> 8).unwrap_or(0),
1092                    u8::try_from(blue >> 8).unwrap_or(0),
1093                    255,
1094                ]),
1095            );
1096        }
1097    }
1098
1099    let mut bytes = Vec::new();
1100    DynamicImage::ImageRgba8(rgba)
1101        .write_to(&mut Cursor::new(&mut bytes), RasterImageFormat::Png)
1102        .map_err(|error| format!("failed to encode captured X11 image as png: {error}"))?;
1103    Ok(bytes)
1104}

1106#[allow(clippy::too_many_arguments, clippy::too_many_lines)]
1107fn dispatch_action(
1108    connection: &X11Connection,
1109    keyboard_map: &KeyboardMap,
1110    request: &InputRequest,
1111    action: &InputAction,
1112    action_index: usize,
1113    label: &str,
1114    held_modifiers: &mut HashSet<ModifierKey>,
1115    notes: &mut Vec<String>,
1116) -> Result<(), TendrilError> {
1117    match action {
1118        InputAction::KeyTap { key } => {
1119            let keysym = key_name_to_keysym(key).ok_or_else(|| {
1120                input_execution_error(
1121                    "unsupported_key",
1122                    format!("key `{key}` is not supported by the X11 backend"),
1123                    Some(action_index),
1124                    Some(label),
1125                )
1126            })?;
1127            let stroke = keyboard_map.stroke_for_keysym(keysym).ok_or_else(|| {
1128                input_execution_error(
1129                    "key_not_mapped",
1130                    format!("key `{key}` is not available in the active X11 keyboard map"),
1131                    Some(action_index),
1132                    Some(label),
1133                )
1134            })?;
1135            send_key_stroke(
1136                connection,
1137                keyboard_map,
1138                stroke,
1139                held_modifiers,
1140                Some(action_index),
1141                Some(label),
1142            )
1143        }
1144        InputAction::Hold { modifier } => {
1145            let stroke = keyboard_map.modifier_stroke(*modifier).ok_or_else(|| {
1146                input_execution_error(
1147                    "modifier_not_mapped",
1148                    format!(
1149                        "modifier `{modifier:?}` is not available in the active X11 keyboard map"
1150                    ),
1151                    Some(action_index),
1152                    Some(label),
1153                )
1154            })?;
1155            fake_key_event(
1156                connection,
1157                KEY_PRESS_EVENT,
1158                stroke.keycode,
1159                Some(action_index),
1160                Some(label),
1161            )?;
1162            held_modifiers.insert(*modifier);
1163            Ok(())
1164        }
1165        InputAction::Release { modifier } => {
1166            let stroke = keyboard_map.modifier_stroke(*modifier).ok_or_else(|| {
1167                input_execution_error(
1168                    "modifier_not_mapped",
1169                    format!(
1170                        "modifier `{modifier:?}` is not available in the active X11 keyboard map"
1171                    ),
1172                    Some(action_index),
1173                    Some(label),
1174                )
1175            })?;
1176            fake_key_event(
1177                connection,
1178                KEY_RELEASE_EVENT,
1179                stroke.keycode,
1180                Some(action_index),
1181                Some(label),
1182            )?;
1183            held_modifiers.remove(modifier);
1184            Ok(())
1185        }
1186        InputAction::Send { text } => {
1187            let dispatch_notes = type_text(
1188                connection,
1189                keyboard_map,
1190                request,
1191                text,
1192                held_modifiers,
1193                Some(action_index),
1194                Some(label),
1195            )?;
1196            notes.extend(dispatch_notes);
1197            Ok(())
1198        }
1199        InputAction::Wait { duration_ms } => {
1200            std::thread::sleep(std::time::Duration::from_millis(*duration_ms));
1201            Ok(())
1202        }
1203        InputAction::Click { button, x, y } => click_button(
1204            connection,
1205            request,
1206            *button,
1207            *x,
1208            *y,
1209            Some(action_index),
1210            Some(label),
1211        ),
1212        InputAction::PointerMove { x, y } => {
1213            pointer_move(connection, request, *x, *y, Some(action_index), Some(label))
1214        }
1215        InputAction::DoubleClick { x, y } => {
1216            double_click_button(connection, request, *x, *y, Some(action_index), Some(label))
1217        }
1218        InputAction::Drag { x0, y0, x1, y1 } => drag_mouse(
1219            connection,
1220            request,
1221            *x0,
1222            *y0,
1223            *x1,
1224            *y1,
1225            Some(action_index),
1226            Some(label),
1227        ),
1228        InputAction::Scroll { x, y, dy } => scroll_wheel(
1229            connection,
1230            request,
1231            *x,
1232            *y,
1233            *dy,
1234            Some(action_index),
1235            Some(label),
1236        ),
1237        InputAction::ElementClick { .. } => Err(input_execution_error(
1238            "unresolved_element_click_action",
1239            "click(<element-id>) should be resolved to target coordinates before reaching the X11 input adapter".to_owned(),
1240            Some(action_index),
1241            Some(label),
1242        )),
1243    }
1244}

1383#[allow(clippy::too_many_arguments)]
1384fn drag_mouse(
1385    connection: &X11Connection,
1386    request: &InputRequest,
1387    x0: i32,
1388    y0: i32,
1389    x1: i32,
1390    y1: i32,
1391    action_index: Option<usize>,
1392    action: Option<&str>,
1393) -> Result<(), TendrilError> {
1394    let (start_x, start_y) = absolute_point(request, x0, y0);
1395    let (end_x, end_y) = absolute_point(request, x1, y1);
1396    move_pointer(connection, start_x, start_y, action_index, action)?;
1397    flush_x11_input(connection, action_index, action, "drag start pointer move")?;
1398    std::thread::sleep(reliability_delay());
1399
1400    fake_button_event(connection, BUTTON_PRESS_EVENT, 1, action_index, action)?;
1401    flush_x11_input(connection, action_index, action, "drag button press")?;
1402    std::thread::sleep(std::time::Duration::from_millis(DRAG_BUTTON_HOLD_DELAY_MS));
1403
1404    for (x, y) in drag_motion_points(start_x, start_y, end_x, end_y) {
1405        move_pointer(connection, x, y, action_index, action)?;
1406        flush_x11_input(connection, action_index, action, "drag motion")?;
1407        std::thread::sleep(reliability_delay());
1408    }
1409
1410    fake_button_event(connection, BUTTON_RELEASE_EVENT, 1, action_index, action)?;
1411    flush_x11_input(connection, action_index, action, "drag button release")
1412}

1414fn drag_motion_points(start_x: i32, start_y: i32, end_x: i32, end_y: i32) -> Vec<(i32, i32)> {
1415    let dx = i64::from(end_x) - i64::from(start_x);
1416    let dy = i64::from(end_y) - i64::from(start_y);
1417    let distance = dx.unsigned_abs().max(dy.unsigned_abs());
1418    let steps = ((distance / DRAG_MOTION_STEP_PX) + 1).clamp(DRAG_MIN_STEPS, DRAG_MAX_STEPS);
1419    (1..=steps)
1420        .map(|step| {
1421            let step = i64::try_from(step).unwrap_or(1);
1422            let steps = i64::try_from(steps).unwrap_or(1);
1423            let x = i64::from(start_x) + (dx * step / steps);
1424            let y = i64::from(start_y) + (dy * step / steps);
1425            (
1426                i32::try_from(x).unwrap_or(if x < 0 { i32::MIN } else { i32::MAX }),
1427                i32::try_from(y).unwrap_or(if y < 0 { i32::MIN } else { i32::MAX }),
1428            )
1429        })
1430        .collect()
1431}

1535fn type_text(
1536    connection: &X11Connection,
1537    keyboard_map: &KeyboardMap,
1538    request: &InputRequest,
1539    text: &str,
1540    held_modifiers: &HashSet<ModifierKey>,
1541    action_index: Option<usize>,
1542    action: Option<&str>,
1543) -> Result<Vec<String>, TendrilError> {
1544    if let Some(character) = first_unmapped_text_character(keyboard_map, text) {
1545        if !held_modifiers.is_empty() {
1546            return Err(input_execution_error(
1547                "character_not_mapped",
1548                format!("character `{character}` is not available in the active X11 keyboard map and clipboard paste fallback cannot run while modifiers are held"),
1549                action_index,
1550                action,
1551            )
1552            .with_detail_entry("fallback", serde_json::json!("x11_clipboard_paste"))
1553            .with_detail_entry("fallback_available", serde_json::json!(false))
1554            .with_detail_entry(
1555                "suggested_action",
1556                serde_json::json!("Release held modifiers before send(...) when entering non-ASCII text on X11, or paste with an explicit clipboard workflow."),
1557            ));
1558        }
1559        if is_x11_temporary_keysym_unicode_target(request) {
1560            return type_text_via_temporary_x11_keysyms(
1561                connection,
1562                keyboard_map,
1563                text,
1564                character,
1565                held_modifiers,
1566                action_index,
1567                action,
1568            );
1569        }
1570        return paste_text_via_x11_selection(
1571            connection,
1572            keyboard_map,
1573            request,
1574            text,
1575            character,
1576            action_index,
1577            action,
1578        );
1579    }
1580
1581    for character in text.chars() {
1582        let stroke = keyboard_map.stroke_for_char(character).ok_or_else(|| {
1583            input_execution_error(
1584                "character_not_mapped",
1585                format!("character `{character}` is not available in the active X11 keyboard map"),
1586                action_index,
1587                action,
1588            )
1589        })?;
1590        send_key_stroke(
1591            connection,
1592            keyboard_map,
1593            stroke,
1594            held_modifiers,
1595            action_index,
1596            action,
1597        )?;
1598    }
1599    Ok(Vec::new())
1600}

1607fn type_text_via_temporary_x11_keysyms(
1608    connection: &X11Connection,
1609    keyboard_map: &KeyboardMap,
1610    text: &str,
1611    unmapped_character: char,
1612    held_modifiers: &HashSet<ModifierKey>,
1613    action_index: Option<usize>,
1614    action: Option<&str>,
1615) -> Result<Vec<String>, TendrilError> {
1616    let mut remapped_count = 0usize;
1617    for character in text.chars() {
1618        if let Some(stroke) = keyboard_map.stroke_for_char(character) {
1619            send_key_stroke(
1620                connection,
1621                keyboard_map,
1622                stroke,
1623                held_modifiers,
1624                action_index,
1625                action,
1626            )?;
1627        } else {
1628            send_temporary_x11_keysym_character(
1629                connection,
1630                keyboard_map,
1631                character,
1632                held_modifiers,
1633                action_index,
1634                action,
1635            )?;
1636            remapped_count += 1;
1637        }
1638    }
1639
1640    Ok(vec![format!(
1641        "X11 send(...) temporarily mapped a keycode for {remapped_count} Unicode character(s) because character `{unmapped_character}` is not present in the active keyboard map; this preserves rich-editor caret state without replacing existing content."
1642    )])
1643}

1645fn send_temporary_x11_keysym_character(
1646    connection: &X11Connection,
1647    keyboard_map: &KeyboardMap,
1648    character: char,
1649    held_modifiers: &HashSet<ModifierKey>,
1650    action_index: Option<usize>,
1651    action: Option<&str>,
1652) -> Result<(), TendrilError> {
1653    let keycode = keyboard_map.temporary_unicode_keycode();
1654    let original = keyboard_map.keysyms_for_keycode(keycode).ok_or_else(|| {
1655        input_execution_error(
1656            "unicode_keycode_not_available",
1657            "no X11 keycode is available for temporary Unicode keysym mapping",
1658            action_index,
1659            action,
1660        )
1661    })?;
1662    let mut replacement = vec![0; usize::from(keyboard_map.keysyms_per_keycode)];
1663    if let Some(first) = replacement.first_mut() {
1664        *first = keysym_for_char(character);
1665    }
1666
1667    change_single_key_mapping(
1668        connection,
1669        keycode,
1670        keyboard_map.keysyms_per_keycode,
1671        &replacement,
1672        action_index,
1673        action,
1674    )?;
1675    std::thread::sleep(std::time::Duration::from_millis(
1676        X11_TEMPORARY_KEYSYM_SETTLE_MS,
1677    ));
1678    let dispatch_result = send_key_stroke(
1679        connection,
1680        keyboard_map,
1681        KeyStroke {
1682            keycode,
1683            shift: false,
1684        },
1685        held_modifiers,
1686        action_index,
1687        action,
1688    )
1689    .and_then(|()| {
1690        // X11 KeyPress events carry a keycode, and clients resolve that
1691        // keycode against the keyboard map when they process the event. Flush
1692        // the synthetic key event and leave the temporary map installed briefly
1693        // before restoring it, otherwise slower consumers such as Firefox's
1694        // contenteditable rich editor can observe the restored map and drop the
1695        // first unmapped Unicode segment.
1696        flush_x11_input(
1697            connection,
1698            action_index,
1699            action,
1700            "temporary Unicode key event",
1701        )?;
1702        std::thread::sleep(std::time::Duration::from_millis(
1703            X11_TEMPORARY_KEYSYM_SETTLE_MS,
1704        ));
1705        Ok(())
1706    });
1707
1708    let restore_result = change_single_key_mapping(
1709        connection,
1710        keycode,
1711        keyboard_map.keysyms_per_keycode,
1712        &original,
1713        action_index,
1714        action,
1715    );
1716    dispatch_result?;
1717    restore_result?;
1718    std::thread::sleep(reliability_delay());
1719    Ok(())
1720}

1722fn change_single_key_mapping(
1723    connection: &X11Connection,
1724    keycode: u8,
1725    keysyms_per_keycode: u8,
1726    keysyms: &[u32],
1727    action_index: Option<usize>,
1728    action: Option<&str>,
1729) -> Result<(), TendrilError> {
1730    connection
1731        .conn
1732        .change_keyboard_mapping(1, keycode, keysyms_per_keycode, keysyms)
1733        .map_err(|error| {
1734            input_execution_error(
1735                "unicode_keycode_mapping_failed",
1736                format!("failed to temporarily change the X11 keyboard mapping: {error}"),
1737                action_index,
1738                action,
1739            )
1740        })?;
1741    flush_x11_input(
1742        connection,
1743        action_index,
1744        action,
1745        "temporary Unicode key mapping",
1746    )
1747}

1772fn paste_text_via_x11_selection(
1773    connection: &X11Connection,
1774    keyboard_map: &KeyboardMap,
1775    request: &InputRequest,
1776    text: &str,
1777    unmapped_character: char,
1778    action_index: Option<usize>,
1779    action: Option<&str>,
1780) -> Result<Vec<String>, TendrilError> {
1781    let attempts = x11_unicode_paste_attempts(request);
1782    let mut failed_attempts = Vec::new();
1783    let mut last_error = None;
1784
1785    for attempt in attempts {
1786        match try_x11_unicode_paste_attempt(
1787            connection,
1788            keyboard_map,
1789            text,
1790            attempt,
1791            action_index,
1792            action,
1793        ) {
1794            Ok(Some((served_requests, mut notes))) => {
1795                let fallback = if is_terminal_paste_attempt(attempt) {
1796                    "terminal-compatible selection paste fallback"
1797                } else {
1798                    "transient CLIPBOARD paste fallback"
1799                };
1800                notes.insert(
1801                    0,
1802                    format!(
1803                        "X11 send(...) used a {fallback} because character `{unmapped_character}` is not present in the active keyboard map; Tendril temporarily owned the {} selection and dispatched {}.",
1804                        clipboard_selection_name(attempt.selection).to_uppercase(),
1805                        attempt.shortcut.label()
1806                    ),
1807                );
1808                notes.push(format!(
1809                    "X11 Unicode paste fallback served {served_requests} selection request(s) for {} bytes of UTF-8 text via {} + {}.",
1810                    text.len(),
1811                    clipboard_selection_name(attempt.selection),
1812                    attempt.shortcut.label()
1813                ));
1814                return Ok(notes);
1815            }
1816            Ok(None) => failed_attempts.push(serde_json::json!({
1817                "selection": clipboard_selection_name(attempt.selection),
1818                "shortcut": attempt.shortcut.label(),
1819                "served_requests": 0,
1820            })),
1821            Err(error) => {
1822                failed_attempts.push(serde_json::json!({
1823                    "selection": clipboard_selection_name(attempt.selection),
1824                    "shortcut": attempt.shortcut.label(),
1825                    "error": error.to_string(),
1826                }));
1827                last_error = Some(error);
1828            }
1829        }
1830    }
1831
1832    let terminal_target = is_x11_terminal_target(request);
1833    let mut error = input_execution_error(
1834        "clipboard_paste_unserved",
1835        if terminal_target {
1836            "X11 Unicode paste fallback tried terminal-compatible paste shortcuts, but no application requested Tendril's selection text"
1837        } else {
1838            "X11 clipboard paste fallback dispatched Ctrl+V but no application requested Tendril's clipboard text"
1839        },
1840        action_index,
1841        action,
1842    )
1843    .with_detail_entry(
1844        "fallback",
1845        serde_json::json!(if terminal_target {
1846            "x11_terminal_selection_paste"
1847        } else {
1848            "x11_clipboard_paste"
1849        }),
1850    )
1851    .with_detail_entry("text_len", serde_json::json!(text.len()))
1852    .with_detail_entry("attempts", serde_json::json!(failed_attempts))
1853    .with_detail_entry(
1854        "suggested_action",
1855        serde_json::json!(if terminal_target {
1856            "Capture the terminal to verify focus is at a shell prompt, then retry send(...). Tendril should use PRIMARY+Shift+Insert for XTerm-like targets and CLIPBOARD terminal paste chords for other X11 terminals."
1857        } else {
1858            "Capture the target to verify focus is in an editable control, then retry send(...) or use tendril clipboard set plus an explicit paste shortcut."
1859        }),
1860    );
1861    if let Some(last_error) = last_error {
1862        error = error.with_detail_entry("last_error", serde_json::json!(last_error.to_string()));
1863    }
1864    Err(error)
1865}

1867fn try_x11_unicode_paste_attempt(
1868    connection: &X11Connection,
1869    keyboard_map: &KeyboardMap,
1870    text: &str,
1871    attempt: X11UnicodePasteAttempt,
1872    action_index: Option<usize>,
1873    action: Option<&str>,
1874) -> Result<Option<(usize, Vec<String>)>, TendrilError> {
1875    let owner = serve_x11_clipboard_in_background(
1876        attempt.selection,
1877        text,
1878        std::time::Duration::from_millis(X11_UNICODE_PASTE_SERVE_MS),
1879    )
1880    .map_err(|error| {
1881        error
1882            .with_detail_entry("fallback", serde_json::json!("x11_unicode_paste"))
1883            .with_detail_entry(
1884                "selection",
1885                serde_json::json!(clipboard_selection_name(attempt.selection)),
1886            )
1887            .with_detail_entry("shortcut", serde_json::json!(attempt.shortcut.label()))
1888            .with_detail_entry("action", serde_json::json!(action))
1889    })?;
1890
1891    let paste_result = send_x11_paste_shortcut(
1892        connection,
1893        keyboard_map,
1894        attempt.shortcut,
1895        action_index,
1896        action,
1897    );
1898    let serve_result = owner.join();
1899    if let Err(error) = paste_result {
1900        let _ = serve_result;
1901        return Err(error
1902            .with_detail_entry(
1903                "selection",
1904                serde_json::json!(clipboard_selection_name(attempt.selection)),
1905            )
1906            .with_detail_entry("shortcut", serde_json::json!(attempt.shortcut.label())));
1907    }
1908    let (served_requests, notes) = serve_result.map_err(|error| {
1909        error
1910            .with_detail_entry("fallback", serde_json::json!("x11_unicode_paste"))
1911            .with_detail_entry(
1912                "selection",
1913                serde_json::json!(clipboard_selection_name(attempt.selection)),
1914            )
1915            .with_detail_entry("shortcut", serde_json::json!(attempt.shortcut.label()))
1916            .with_detail_entry("action", serde_json::json!(action))
1917    })?;
1918
1919    if served_requests == 0 {
1920        Ok(None)
1921    } else {
1922        Ok(Some((served_requests, notes)))
1923    }
1924}

2461struct KeyboardMap {
2462    min_keycode: u8,
2463    max_keycode: u8,
2464    keysyms_per_keycode: u8,
2465    keysyms: Vec<u32>,
2466}
2467
2468impl KeyboardMap {
2469    fn load(connection: &X11Connection) -> Result<Self, String> {
2470        let min_keycode = connection.conn.setup().min_keycode;
2471        let max_keycode = connection.conn.setup().max_keycode;
2472        let count = max_keycode.saturating_sub(min_keycode).saturating_add(1);
2473        let reply = connection
2474            .conn
2475            .get_keyboard_mapping(min_keycode, count)
2476            .map_err(|error| error.to_string())?
2477            .reply()
2478            .map_err(|error| error.to_string())?;
2479        Ok(Self {
2480            min_keycode,
2481            max_keycode,
2482            keysyms_per_keycode: reply.keysyms_per_keycode,
2483            keysyms: reply.keysyms,
2484        })
2485    }
2486
2487    fn modifier_stroke(&self, modifier: ModifierKey) -> Option<KeyStroke> {
2488        let keysym = match modifier {
2489            ModifierKey::Ctrl => XK_CONTROL_L,
2490            ModifierKey::Alt => XK_ALT_L,
2491            ModifierKey::Shift => XK_SHIFT_L,
2492            ModifierKey::Meta => XK_SUPER_L,
2493        };
2494        self.stroke_for_keysym(keysym)
2495    }
2496
2497    fn stroke_for_char(&self, character: char) -> Option<KeyStroke> {
2498        self.stroke_for_keysym(keysym_for_char(character))
2499    }
2500
2501    fn stroke_for_keysym(&self, keysym: u32) -> Option<KeyStroke> {
2502        for keycode in self.min_keycode..=self.max_keycode {
2503            let keysyms = self.keysyms_for_keycode(keycode)?;
2504            if keysyms.first().copied() == Some(keysym) {
2505                return Some(KeyStroke {
2506                    keycode,
2507                    shift: false,
2508                });
2509            }
2510            if keysyms.get(1).copied() == Some(keysym) {
2511                return Some(KeyStroke {
2512                    keycode,
2513                    shift: true,
2514                });
2515            }
2516            if keysyms.iter().copied().any(|candidate| candidate == keysym) {
2517                return Some(KeyStroke {
2518                    keycode,
2519                    shift: false,
2520                });
2521            }
2522        }
2523        None
2524    }

2526    fn keysyms_for_keycode(&self, keycode: u8) -> Option<Vec<u32>> {
2527        if keycode < self.min_keycode || keycode > self.max_keycode {
2528            return None;
2529        }
2530        let per_keycode = usize::from(self.keysyms_per_keycode);
2531        let start = usize::from(keycode.saturating_sub(self.min_keycode)) * per_keycode;
2532        self.keysyms
2533            .get(start..start + per_keycode)
2534            .map(<[u32]>::to_vec)
2535    }
2536
2537    fn temporary_unicode_keycode(&self) -> u8 {
2538        (self.min_keycode..=self.max_keycode)
2539            .rev()
2540            .find(|keycode| {
2541                self.keysyms_for_keycode(*keycode)
2542                    .is_some_and(|keysyms| keysyms.iter().all(|keysym| *keysym == 0))
2543            })
2544            .unwrap_or(self.max_keycode)
2545    }
2546}

2684        .split(|byte| *byte == 0)
2685        .filter_map(|value| {
2686            let value = String::from_utf8_lossy(value).trim().to_owned();
2687            (!value.is_empty()).then_some(value)
2688        })
2689        .collect::<Vec<_>>();
2690    values.last().cloned()
2691}
