1//! Wayland input injection backend.
2//!
3//! This module powers `tendril run` on Wayland sessions (Hyprland, sway, and
4//! other wlroots-based compositors) by delegating actual event delivery to two
5//! widely available helper tools:
6//!
7//! * `ydotool` — kernel-level input injection via `/dev/uinput` driven by a
8//!   `ydotoold` daemon. Preferred because it works across every Wayland
9//!   compositor and supports both keyboard and pointer events.
10//! * `wtype` — keyboard input injection through the wlroots-specific
11//!   `virtual-keyboard-v1` Wayland protocol. Used as a keyboard-only fallback
12//!   when `ydotool` is not present.
13//!
14//! Tools are detected at runtime; the public [`detect_backend`] helper returns
15//! a [`WaylandInputCapability`] that other modules use to decide whether
16//! Wayland targets should advertise `input_supported = true`.
17

33/// Names of the helper binaries that this module knows how to drive.
34const YDOTOOL_BIN: &str = "ydotool";
35const WTYPE_BIN: &str = "wtype";
36
37/// Linux input event "press" / "release" values. Mirrors `linux/input-event-codes.h`.
38const KEY_DOWN: u32 = 1;
39const KEY_UP: u32 = 0;
40
41/// Picks which Wayland helper tool will service a given request.
42#[derive(Debug, Clone, Copy, PartialEq, Eq)]
43pub(crate) enum WaylandInputBackend {
44    /// `ydotool` + `ydotoold`; covers keyboard and pointer events.
45    Ydotool,
46    /// `wtype`; covers keyboard events only (no pointer support).
47    Wtype,
48}
49
50/// Snapshot of the Wayland input helpers detected on the current PATH.
51#[derive(Debug, Clone, Default, PartialEq, Eq)]
52pub(crate) struct WaylandInputCapability {
53    pub ydotool: bool,
54    pub wtype: bool,
55}
56
57impl WaylandInputCapability {
58    /// Returns true when at least one helper that can deliver Wayland input is
59    /// available. Even a `wtype`-only host can service keyboard sequences, so
60    /// we still advertise input support and surface a clear runtime error if a
61    /// caller asks for mouse events that the chosen backend cannot handle.
62    pub(crate) fn any_supported(&self) -> bool {
63        self.ydotool || self.wtype
64    }
65
66    /// Backend that should be preferred when both helpers are present.
67    pub(crate) fn preferred(&self) -> Option<WaylandInputBackend> {
68        if self.ydotool {
69            Some(WaylandInputBackend::Ydotool)
70        } else if self.wtype {
71            Some(WaylandInputBackend::Wtype)
72        } else {
73            None
74        }
75    }
76}

78/// Detects which Wayland input helper tools are reachable on `PATH`.
79pub(crate) fn detect_backend() -> WaylandInputCapability {
80    WaylandInputCapability {
81        ydotool: program_on_path(YDOTOOL_BIN),
82        wtype: program_on_path(WTYPE_BIN),
83    }
84}

86/// Returns the diagnostic surfaced by `tendril run` when no Wayland helper is
87/// available. Centralised here so the [`crate::platform::LinuxAdapter`] and
88/// the runtime executor share identical guidance.
89pub(crate) fn missing_backend_error(platform: PlatformKind) -> PlatformAdapterError {
90    PlatformAdapterError::unsupported(
91        Capability::InputControl,
92        platform,
93        CapabilityErrorReason::UnsupportedFeature,
94        "Wayland input injection requires either `ydotool` (with the `ydotoold` daemon) or `wtype` to be installed and reachable on PATH.",
95        Some(
96            "Install `ydotool` for full keyboard + pointer support (preferred), or install `wtype` for keyboard-only input on wlroots-based compositors such as Hyprland and sway. Make sure the `ydotoold` daemon is running and that the invoking user can reach its socket (typically `/tmp/.ydotool_socket` or `$YDOTOOL_SOCKET`).",
97        ),
98    )
99}

101/// Executes the requested input sequence using a Wayland helper tool.
102pub(crate) fn execute_input(
103    platform: PlatformKind,
104    request: &InputRequest,
105) -> Result<InputOutcome, TendrilError> {
106    let capability = detect_backend();
107    let Some(backend) = capability.preferred() else {
108        return Err(TendrilError::from(missing_backend_error(platform)));
109    };
110
111    if backend == WaylandInputBackend::Wtype && request_has_pointer(request) {
112        // wtype cannot synthesize pointer events. Surface an actionable
113        // diagnostic instead of silently dropping the click/drag.
114        return Err(TendrilError::from(PlatformAdapterError::unsupported(
115            Capability::InputControl,
116            platform,
117            CapabilityErrorReason::UnsupportedFeature,
118            "The detected Wayland input backend `wtype` only supports keyboard events; the requested sequence contains pointer events.",
119            Some(
120                "Install `ydotool` (and run the `ydotoold` daemon) to enable pointer events on Wayland sessions. wtype intentionally targets the wlroots virtual-keyboard protocol and does not implement pointer injection.",
121            ),
122        )));
123    }
124
125    let mut notes = Vec::new();
126    notes.push(match backend {
127        WaylandInputBackend::Ydotool => {
128            "Wayland input dispatched through ydotool (uinput); requires the ydotoold daemon and PATH access to the helper binary."
129                .to_owned()
130        }
131        WaylandInputBackend::Wtype => {
132            "Wayland input dispatched through wtype using the wlroots virtual-keyboard-v1 protocol; pointer events are not supported by this backend."
133                .to_owned()
134        }
135    });
136
137    let keyboard_input = request.text.is_some() || request.actions.iter().any(action_is_keyboard);
138    let focus_required = keyboard_input;
139
140    // Wayland compositors do not expose a portable "focus this client"
141    // primitive that works for arbitrary surfaces, so we record the focus
142    // contract honestly: callers must place focus themselves (or rely on
143    // compositor-side keyboard routing) before issuing the run.
144    if keyboard_input {
145        notes.push(
146            "Wayland focus transfer is compositor-mediated; place focus on the intended surface before running keyboard sequences for reliable delivery."
147                .to_owned(),
148        );
149    }
150
151    if let Some(text) = &request.text {
152        dispatch_text(backend, text, Some(0), Some("text"))?;
153        if request.restore_focus {
154            notes.push(
155                "Wayland focus restoration is not portable through Tendril; no previous focus snapshot was captured."
156                    .to_owned(),
157            );
158        }
159        return Ok(InputOutcome {
160            action_count: 1,
161            focus_required,
162            focus_transferred: false,
163            focused_target: None,
164            previous_focus: None,
165            focus_restored: false,
166            pointer_restored: false,
167            restore_error: request.restore_focus.then(|| {
168                "focus restoration is not supported for generic Wayland sessions".to_owned()
169            }),
170            notes,
171        });
172    }
173
174    let mut held_modifiers: HashSet<ModifierKey> = HashSet::new();
175    for (action_index, action) in request.actions.iter().enumerate() {
176        let label = action_label(action);
177        dispatch_action(
178            backend,
179            request,
180            action,
181            action_index,
182            &label,
183            &mut held_modifiers,
184        )?;
185        if !matches!(action, InputAction::Wait { .. }) {
186            std::thread::sleep(reliability_delay());
187        }
188    }
189
190    if request.restore_focus {
191        notes.push(
192            "Wayland focus restoration is not portable through Tendril; no previous focus snapshot was captured."
193                .to_owned(),
194        );
195    }
196
197    Ok(InputOutcome {
198        action_count: request.actions.len(),
199        focus_required,
200        focus_transferred: false,
201        focused_target: None,
202        previous_focus: None,
203        focus_restored: false,
204        pointer_restored: false,
205        restore_error: request
206            .restore_focus
207            .then(|| "focus restoration is not supported for generic Wayland sessions".to_owned()),
208        notes,
209    })
210}

212fn dispatch_action(
213    backend: WaylandInputBackend,
214    request: &InputRequest,
215    action: &InputAction,
216    action_index: usize,
217    label: &str,
218    held_modifiers: &mut HashSet<ModifierKey>,
219) -> Result<(), TendrilError> {
220    match action {
221        InputAction::KeyTap { key } => dispatch_key_tap(
222            backend,
223            key,
224            held_modifiers,
225            Some(action_index),
226            Some(label),
227        ),
228        InputAction::Hold { modifier } => {
229            dispatch_modifier(backend, *modifier, true, Some(action_index), Some(label))?;
230            held_modifiers.insert(*modifier);
231            Ok(())
232        }
233        InputAction::Release { modifier } => {
234            dispatch_modifier(backend, *modifier, false, Some(action_index), Some(label))?;
235            held_modifiers.remove(modifier);
236            Ok(())
237        }
238        InputAction::Send { text } => dispatch_text(backend, text, Some(action_index), Some(label)),
239        InputAction::Wait { duration_ms } => {
240            std::thread::sleep(Duration::from_millis(*duration_ms));
241            Ok(())
242        }
243        InputAction::Click { button, x, y } => {
244            let (absolute_x, absolute_y) = relative_point_to_absolute(&request.bounds, *x, *y);
245            dispatch_click(
246                backend,
247                *button,
248                absolute_x,
249                absolute_y,
250                Some(action_index),
251                Some(label),
252            )
253        }
254        InputAction::PointerMove { .. } => Err(input_execution_error(
255            "unsupported_pointer_move_action",
256            "move(...) and hover(...) are currently implemented for Linux/X11 input delivery; this Wayland adapter does not yet support native pointer-only motion".to_owned(),
257            "dispatch",
258            Some(action_index),
259            Some(label),
260        )),
261        InputAction::DoubleClick { .. } => Err(input_execution_error(
262            "unsupported_double_click_action",
263            "dblclick(...) is currently implemented for Linux/X11 input delivery; this Wayland adapter does not yet support native double-click injection".to_owned(),
264            "dispatch",
265            Some(action_index),
266            Some(label),
267        )),
268        InputAction::Drag { x0, y0, x1, y1 } => {
269            let (start_x, start_y) = relative_point_to_absolute(&request.bounds, *x0, *y0);
270            let (end_x, end_y) = relative_point_to_absolute(&request.bounds, *x1, *y1);
271            dispatch_drag(
272                backend,
273                start_x,
274                start_y,
275                end_x,
276                end_y,
277                Some(action_index),
278                Some(label),
279            )
280        }
281        InputAction::Scroll { .. } => Err(input_execution_error(
282            "unsupported_scroll_action",
283            "scroll(...) is currently implemented for Linux/X11 input delivery; this Wayland adapter does not yet support native wheel injection".to_owned(),
284            "dispatch",
285            Some(action_index),
286            Some(label),
287        )),
288        InputAction::ElementClick { .. } => Err(input_execution_error(
289            "unresolved_element_click_action",
290            "click(<element-id>) should be resolved to target coordinates before reaching the Wayland input adapter".to_owned(),
291            "dispatch",
292            Some(action_index),
293            Some(label),
294        )),
295    }
296}

298fn dispatch_text(
299    backend: WaylandInputBackend,
300    text: &str,
301    action_index: Option<usize>,
302    action: Option<&str>,
303) -> Result<(), TendrilError> {
304    if text.is_empty() {
305        return Ok(());
306    }
307
308    match backend {
309        WaylandInputBackend::Ydotool => run_helper(
310            YDOTOOL_BIN,
311            &["type", "--", text],
312            "dispatch",
313            action_index,
314            action,
315        ),
316        WaylandInputBackend::Wtype => {
317            // `wtype --` followed by the literal text avoids any further flag parsing.
318            run_helper(WTYPE_BIN, &["--", text], "dispatch", action_index, action)
319        }
320    }
321}

323fn dispatch_key_tap(
324    backend: WaylandInputBackend,
325    key: &str,
326    held_modifiers: &HashSet<ModifierKey>,
327    action_index: Option<usize>,
328    action: Option<&str>,
329) -> Result<(), TendrilError> {
330    match backend {
331        WaylandInputBackend::Ydotool => {
332            let keycode = ydotool_key_code(key).ok_or_else(|| {
333                input_execution_error(
334                    "unsupported_key",
335                    format!("key `{key}` is not supported by the ydotool Wayland backend"),
336                    "dispatch",
337                    action_index,
338                    action,
339                )
340            })?;
341            let press = format!("{keycode}:{KEY_DOWN}");
342            let release = format!("{keycode}:{KEY_UP}");
343            run_helper(
344                YDOTOOL_BIN,
345                &["key", "--", press.as_str(), release.as_str()],
346                "dispatch",
347                action_index,
348                action,
349            )
350        }
351        WaylandInputBackend::Wtype => {
352            let mut args: Vec<String> = Vec::new();
353            // Translate any persistently held modifiers into wtype's per-invocation
354            // modifier flags (`-M ctrl ... -m ctrl`).
355            for modifier in held_modifiers {
356                args.push("-M".to_owned());
357                args.push(wtype_modifier_name(*modifier).to_owned());
358            }
359            let key_argument = wtype_key_argument(key).ok_or_else(|| {
360                input_execution_error(
361                    "unsupported_key",
362                    format!("key `{key}` is not supported by the wtype Wayland backend"),
363                    "dispatch",
364                    action_index,
365                    action,
366                )
367            })?;
368            args.push("-k".to_owned());
369            args.push(key_argument);
370            for modifier in held_modifiers {
371                args.push("-m".to_owned());
372                args.push(wtype_modifier_name(*modifier).to_owned());
373            }
374            let arg_refs: Vec<&str> = args.iter().map(String::as_str).collect();
375            run_helper(WTYPE_BIN, &arg_refs, "dispatch", action_index, action)
376        }
377    }
378}

380fn dispatch_modifier(
381    backend: WaylandInputBackend,
382    modifier: ModifierKey,
383    press: bool,
384    action_index: Option<usize>,
385    action: Option<&str>,
386) -> Result<(), TendrilError> {
387    match backend {
388        WaylandInputBackend::Ydotool => {
389            let keycode = ydotool_modifier_code(modifier);
390            let value = if press { KEY_DOWN } else { KEY_UP };
391            let argument = format!("{keycode}:{value}");
392            run_helper(
393                YDOTOOL_BIN,
394                &["key", "--", argument.as_str()],
395                "dispatch",
396                action_index,
397                action,
398            )
399        }
400        WaylandInputBackend::Wtype => {
401            let modifier_name = wtype_modifier_name(modifier);
402            let direction = if press { "-M" } else { "-m" };
403            run_helper(
404                WTYPE_BIN,
405                &[direction, modifier_name],
406                "dispatch",
407                action_index,
408                action,
409            )
410        }
411    }
412}

414fn dispatch_click(
415    backend: WaylandInputBackend,
416    button: MouseButton,
417    x: i32,
418    y: i32,
419    action_index: Option<usize>,
420    action: Option<&str>,
421) -> Result<(), TendrilError> {
422    debug_assert!(matches!(backend, WaylandInputBackend::Ydotool));
423
424    move_pointer_absolute(x, y, action_index, action)?;
425    // ydotool's `click` argument is a bitmask: low nibble = press mask,
426    // high nibble = release mask. `0x40 | 0x4 = 0x44` ⇒ press+release of the
427    // requested button (0x1=left, 0x2=right, 0x4=middle in the upstream tool).
428    let mask = ydotool_button_combined_mask(button);
429    let mask_argument = format!("0x{mask:X}");
430    run_helper(
431        YDOTOOL_BIN,
432        &["click", "--", mask_argument.as_str()],
433        "dispatch",
434        action_index,
435        action,
436    )
437}

439fn dispatch_drag(
440    backend: WaylandInputBackend,
441    start_x: i32,
442    start_y: i32,
443    end_x: i32,
444    end_y: i32,
445    action_index: Option<usize>,
446    action: Option<&str>,
447) -> Result<(), TendrilError> {
448    debug_assert!(matches!(backend, WaylandInputBackend::Ydotool));
449
450    let press_mask = format!("0x{:X}", ydotool_button_press_mask(MouseButton::Left));
451    let release_mask = format!("0x{:X}", ydotool_button_release_mask(MouseButton::Left));
452
453    move_pointer_absolute(start_x, start_y, action_index, action)?;
454    run_helper(
455        YDOTOOL_BIN,
456        &["click", "--", press_mask.as_str()],
457        "dispatch",
458        action_index,
459        action,
460    )?;
461    std::thread::sleep(reliability_delay());
462    move_pointer_absolute(end_x, end_y, action_index, action)?;
463    run_helper(
464        YDOTOOL_BIN,
465        &["click", "--", release_mask.as_str()],
466        "dispatch",
467        action_index,
468        action,
469    )
470}

472fn move_pointer_absolute(
473    x: i32,
474    y: i32,
475    action_index: Option<usize>,
476    action: Option<&str>,
477) -> Result<(), TendrilError> {
478    // ydotool 1.x interprets `mousemove --absolute -x X -y Y` as global
479    // workspace coordinates, which is what the InputRequest already supplies
480    // after `relative_point_to_absolute` translation.
481    let x_str = x.to_string();
482    let y_str = y.to_string();
483    run_helper(
484        YDOTOOL_BIN,
485        &[
486            "mousemove",
487            "--absolute",
488            "-x",
489            x_str.as_str(),
490            "-y",
491            y_str.as_str(),
492        ],
493        "dispatch",
494        action_index,
495        action,
496    )
497}

499fn run_helper(
500    program: &str,
501    args: &[&str],
502    stage: &'static str,
503    action_index: Option<usize>,
504    action: Option<&str>,
505) -> Result<(), TendrilError> {
506    let output = Command::new(program).args(args).output().map_err(|error| {
507        input_execution_error(
508            "input_spawn_failed",
509            format!("failed to spawn `{program}`: {error}"),
510            stage,
511            action_index,
512            action,
513        )
514    })?;
515
516    if output.status.success() {
517        return Ok(());
518    }
519
520    let stderr = String::from_utf8_lossy(&output.stderr).trim().to_owned();
521    let stdout = String::from_utf8_lossy(&output.stdout).trim().to_owned();
522    let combined = if stderr.is_empty() && stdout.is_empty() {
523        format!("`{program}` exited with status {}", output.status)
524    } else if stderr.is_empty() {
525        format!("`{program}` failed: {stdout}")
526    } else if stdout.is_empty() {
527        format!("`{program}` failed: {stderr}")
528    } else {
529        format!("`{program}` failed: {stdout} | {stderr}")
530    };
531
532    if program == YDOTOOL_BIN && looks_like_ydotool_socket_error(&stderr, &stdout) {
533        return Err(input_execution_error(
534            "ydotoold_unavailable",
535            format!(
536                "{combined} — ydotool requires the `ydotoold` daemon to be running and reachable via its socket (set `YDOTOOL_SOCKET` or run the daemon under the invoking user)."
537            ),
538            stage,
539            action_index,
540            action,
541        ));
542    }
543
544    Err(input_execution_error(
545        "input_command_failed",
546        combined,
547        stage,
548        action_index,
549        action,
550    ))
551}

553fn looks_like_ydotool_socket_error(stderr: &str, stdout: &str) -> bool {
554    let haystack = format!("{stderr} {stdout}").to_ascii_lowercase();
555    haystack.contains("socket")
556        || haystack.contains("connection refused")
557        || haystack.contains("no such file")
558        || haystack.contains("ydotoold")
559        || haystack.contains("permission denied")
560}

562fn request_has_pointer(request: &InputRequest) -> bool {
563    request.actions.iter().any(|action| {
564        matches!(
565            action,
566            InputAction::Click { .. }
567                | InputAction::PointerMove { .. }
568                | InputAction::DoubleClick { .. }
569                | InputAction::Drag { .. }
570                | InputAction::Scroll { .. }
571                | InputAction::ElementClick { .. }

590fn input_execution_error(
591    code: &'static str,
592    message: String,
593    stage: &'static str,
594    action_index: Option<usize>,
595    action: Option<&str>,
596) -> TendrilError {
597    let mut error = TendrilError::execution_failure(code, message, action_index)
598        .with_detail_entry("stage", json!(stage));
599    if let Some(action_index) = action_index {
600        error = error.with_detail_entry("action_number", json!(action_index + 1));
601    }
602    if let Some(action) = action {
603        error = error.with_detail_entry("action", json!(action));
604    }
605    error
606}

608fn program_on_path(program: &str) -> bool {
609    env::var_os("PATH").is_some_and(|path| {
610        env::split_paths(&path).any(|entry| {
611            let candidate = entry.join(program);
612            candidate.is_file()
613        })
614    })
615}

617/// Maps a Tendril modifier into the upstream Linux input event keycode used by
618/// `ydotool key`. The "left" variant is chosen because it is the most widely
619/// honored modifier across compositors when synthesising shortcuts.
620fn ydotool_modifier_code(modifier: ModifierKey) -> u32 {
621    match modifier {
622        ModifierKey::Ctrl => 29,  // KEY_LEFTCTRL
623        ModifierKey::Shift => 42, // KEY_LEFTSHIFT
624        ModifierKey::Alt => 56,   // KEY_LEFTALT
625        ModifierKey::Meta => 125, // KEY_LEFTMETA
626    }
627}

629/// Maps a Tendril modifier into the wtype modifier name (`ctrl`, `shift`,
630/// `alt`, `logo`).
631fn wtype_modifier_name(modifier: ModifierKey) -> &'static str {
632    match modifier {
633        ModifierKey::Ctrl => "ctrl",
634        ModifierKey::Shift => "shift",
635        ModifierKey::Alt => "alt",
636        ModifierKey::Meta => "logo",
637    }
638}

640/// Translates a key token from the Tendril DSL into the literal argument that
641/// `wtype -k` accepts (an XKB keysym name). Returns `None` for unsupported
642/// tokens so callers can surface an actionable error.
643fn wtype_key_argument(key: &str) -> Option<String> {
644    let lower = key.to_ascii_lowercase();
645    match lower.as_str() {
646        "enter" | "return" => Some("Return".to_owned()),
647        "tab" => Some("Tab".to_owned()),
648        "esc" | "escape" => Some("Escape".to_owned()),
649        "space" => Some("space".to_owned()),
650        "backspace" => Some("BackSpace".to_owned()),
651        "delete" | "del" => Some("Delete".to_owned()),
652        "left" => Some("Left".to_owned()),
653        "right" => Some("Right".to_owned()),
654        "up" => Some("Up".to_owned()),
655        "down" => Some("Down".to_owned()),
656        "home" => Some("Home".to_owned()),
657        "end" => Some("End".to_owned()),
658        "pageup" => Some("Prior".to_owned()),
659        "pagedown" => Some("Next".to_owned()),
660        other if other.chars().count() == 1 => other.chars().next().map(|c| c.to_string()),
661        _ => None,
662    }
663}

665/// Maps a key token from the Tendril DSL onto the Linux input event keycode
666/// used by `ydotool key`. Returns `None` for unsupported tokens.
667fn ydotool_key_code(key: &str) -> Option<u32> {

748/// Returns the `ydotool click` mask that performs a press immediately followed
749/// by a release of `button` (high nibble = release, low nibble = press).
750fn ydotool_button_combined_mask(button: MouseButton) -> u32 {
751    let press = ydotool_button_press_mask(button);
752    let release = ydotool_button_release_mask(button);
753    press | release
754}
755
756fn ydotool_button_press_mask(button: MouseButton) -> u32 {
757    match button {
758        MouseButton::Left => 0x40,
759        MouseButton::Right => 0x41,
760        MouseButton::Middle => 0x42,
761    }
762}
763
764fn ydotool_button_release_mask(button: MouseButton) -> u32 {
765    match button {
766        MouseButton::Left => 0x80,
767        MouseButton::Right => 0x81,
768        MouseButton::Middle => 0x82,
769    }
770}
