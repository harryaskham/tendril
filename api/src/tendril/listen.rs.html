1//! Audio capture execution for the `tendril listen` command.
2//!
3//! The v0.0.1 listen surface only reported probe-only diagnostics. This module
4//! drives an actual recording for at least one supported source per platform
5//! by shelling out to a well-known recorder:
6//!
7//! * Linux + `PipeWire`: `pw-record` (with `parecord` fallback).
8//! * Linux + `PulseAudio`: `parecord`.
9//! * macOS: `afrecord` (Apple's CoreAudio-backed recorder shipped with the OS).
10//! * Windows / unknown backends: capture is not yet wired; callers receive a
11//!   structured `probe_only` response.
12//!
13//! Each successful recording writes WAV bytes to either an explicit
14//! `--output` path or a temporary file, and the path is reported back to the
15//! caller in the JSON envelope so downstream agents can fetch the artifact.
16

30/// Outcome of a real audio capture attempt.
31#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
32pub struct ListenArtifact {
33    /// Absolute path to the on-disk WAV file containing the captured audio.
34    pub path: PathBuf,
35    /// MIME type of the artifact (currently always `audio/wav`).
36    pub media_type: String,
37    /// Number of bytes written to disk.
38    pub byte_size: u64,
39    /// Negotiated sample rate in Hz.
40    pub sample_rate_hz: u32,
41    /// Negotiated channel count.
42    pub channels: u8,
43    /// Recorder program that produced the artifact (e.g. `pw-record`).
44    pub recorder: String,
45}
46
47/// Reason a capture attempt was skipped before any recorder was spawned.
48#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
49#[serde(rename_all = "snake_case")]
50pub enum ListenSkipReason {
51    /// No supported recorder was available on this platform.
52    UnsupportedPlatform,
53    /// Requested format is not yet supported by the selected recorder.
54    UnsupportedFormat,
55    /// Requested source kind is not yet supported by the selected recorder.
56    UnsupportedSource,
57    /// Recorder binary could not be located on PATH.
58    RecorderUnavailable,
59}

61/// Result of attempting to perform an actual audio capture.
62#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
63#[serde(tag = "outcome", rename_all = "snake_case")]
64pub enum ListenCaptureResult {
65    /// A real WAV recording was written to disk.
66    Captured {
67        artifact: ListenArtifact,
68        notes: Vec<String>,
69    },
70    /// No recorder ran; the legacy probe-only response stands.
71    Skipped {
72        reason: ListenSkipReason,
73        notes: Vec<String>,
74    },
75    /// A recorder ran but failed; the caller still gets the probe diagnostics.
76    Failed {
77        recorder: String,
78        message: String,
79        notes: Vec<String>,
80    },
81}
82
83impl ListenCaptureResult {
84    #[must_use]
85    pub fn artifact(&self) -> Option<&ListenArtifact> {
86        match self {
87            Self::Captured { artifact, .. } => Some(artifact),
88            _ => None,
89        }
90    }
91
92    #[must_use]
93    pub fn notes(&self) -> &[String] {
94        match self {
95            Self::Captured { notes, .. }
96            | Self::Skipped { notes, .. }
97            | Self::Failed { notes, .. } => notes,
98        }
99    }
100}

102/// Drive a real audio capture for `input` on the active platform.
103///
104/// Errors only when validation of the user-provided `output` path fails
105/// (parent directory missing, etc.). Recorder-level failures are reported as
106/// [`ListenCaptureResult::Failed`] so the caller can still surface probe data.
107pub fn execute_listen_capture(
108    input: &ListenInput,
109    output: Option<&Path>,
110    platform: PlatformKind,
111    backend: Option<AudioBackend>,
112) -> Result<ListenCaptureResult, TendrilError> {
113    if input.format != AudioFormat::Wav {
114        return Ok(ListenCaptureResult::Skipped {
115            reason: ListenSkipReason::UnsupportedFormat,
116            notes: vec![format!(
117                "listen capture currently emits only WAV; requested {:?} format is not yet implemented.",
118                input.format
119            )],
120        });
121    }
122
123    if matches!(input.source.kind, AudioSourceKind::Device) {
124        return Ok(ListenCaptureResult::Skipped {
125            reason: ListenSkipReason::UnsupportedSource,
126            notes: vec![
127                "explicit device:<id> selection is not implemented for real capture; pass --source microphone or --source system instead.".to_owned(),
128            ]
129        });
130    }
131
132    let recorders = recorders_for(platform, backend);
133    if recorders.is_empty() {
134        return Ok(ListenCaptureResult::Skipped {
135            reason: ListenSkipReason::UnsupportedPlatform,
136            notes: vec![format!(
137                "no audio recorder is wired for platform={:?}, backend={:?}; capture remains probe-only here.",
138                platform, backend
139            )],
140        });
141    }
142
143    let target_path = resolve_output_path(output)?;
144
145    let mut last_failure: Option<(String, String)> = None;
146    let mut tried: Vec<String> = Vec::new();
147    for recorder in recorders {
148        if !command_available(recorder.program) {
149            tried.push(format!("{} (not found on PATH)", recorder.program));
150            continue;
151        }
152        tried.push(recorder.program.to_owned());
153        match run_recorder(&recorder, input, &target_path) {
154            Ok(artifact) => {
155                let mut notes = Vec::new();
156                notes.push(format!(
157                    "captured {} bytes to {} via {}",
158                    artifact.byte_size,
159                    artifact.path.display(),
160                    artifact.recorder
161                ));
162                if artifact.byte_size <= WAV_HEADER_ONLY_BYTES {
163                    notes.push(format!(
164                        "warning: artifact contains only WAV header bytes ({} <= {}); the source likely produced no samples (suspended monitor, muted mic, or no audio playing).",
165                        artifact.byte_size, WAV_HEADER_ONLY_BYTES
166                    ));
167                }
168                if tried.len() > 1 {
169                    notes.push(format!(
170                        "recorder candidates considered: {}",
171                        tried.join(", ")
172                    ));
173                }
174                return Ok(ListenCaptureResult::Captured { artifact, notes });
175            }
176            Err(error) => {
177                last_failure = Some((recorder.program.to_owned(), error));
178            }
179        }
180    }
181
182    match last_failure {
183        Some((recorder, message)) => Ok(ListenCaptureResult::Failed {
184            recorder,
185            message,
186            notes: vec![format!(
187                "all recorder attempts failed; tried: {}",
188                tried.join(", ")
189            )],
190        }),
191        None => Ok(ListenCaptureResult::Skipped {
192            reason: ListenSkipReason::RecorderUnavailable,
193            notes: vec![format!(
194                "no usable recorder found; looked for: {}",
195                tried.join(", ")
196            )],
197        }),
198    }
199}

212/// A virtual / aggregate audio input device usable as a system-audio loopback
213/// (e.g. `BlackHole`). macOS has no built-in system loopback, so `--source
214/// system` captures from such a device once the user routes system output to it.
215#[derive(Debug, Clone, PartialEq, Eq)]
216pub(crate) struct AvfAudioDevice {
217    pub index: u32,
218    pub name: String,
219}

221#[cfg(any(target_os = "macos", test))]
222const LOOPBACK_NAME_HINTS: &[&str] = &[
223    "blackhole",
224    "loopback",
225    "soundflower",
226    "aggregate",
227    "multi-output",
228    "multi output",
229];

231/// Parse the `AVFoundation` audio device list ffmpeg prints to stderr for
232/// `ffmpeg -f avfoundation -list_devices true -i ""`. Device lines look like
233/// `[AVFoundation indev @ 0x..] [1] BlackHole 2ch`, under an
234/// `AVFoundation audio devices:` header (the video section is ignored).
235#[cfg(any(target_os = "macos", test))]
236pub(crate) fn parse_avfoundation_audio_devices(stderr: &str) -> Vec<AvfAudioDevice> {
237    let mut devices = Vec::new();
238    let mut in_audio = false;
239    for line in stderr.lines() {
240        if line.contains("AVFoundation audio devices:") {
241            in_audio = true;
242            continue;
243        }
244        if line.contains("AVFoundation video devices:") {
245            in_audio = false;
246            continue;
247        }
248        if !in_audio {
249            continue;
250        }
251        // Skip the `[AVFoundation indev @ 0x..] ` tag, then read `[N] Name`.
252        let Some(tag_end) = line.rfind("] [") else {
253            continue;
254        };
255        let rest = &line[tag_end + 2..];
256        let Some(close) = rest.find(']') else {
257            continue;
258        };
259        let Ok(index) = rest[1..close].trim().parse::<u32>() else {
260            continue;
261        };
262        let name = rest[close + 1..].trim().to_owned();
263        if name.is_empty() {
264            continue;
265        }
266        devices.push(AvfAudioDevice { index, name });
267    }
268    devices
269}

271/// Pick the first device whose name looks like a virtual loopback device.
272#[cfg(any(target_os = "macos", test))]
273pub(crate) fn find_loopback_device(devices: &[AvfAudioDevice]) -> Option<AvfAudioDevice> {
274    devices
275        .iter()
276        .find(|device| {
277            let lower = device.name.to_lowercase();
278            LOOPBACK_NAME_HINTS.iter().any(|hint| lower.contains(hint))
279        })
280        .cloned()
281}

283/// Detect an available virtual loopback audio device on macOS via ffmpeg.
284/// Returns `None` off macOS, when ffmpeg is unavailable, or when no loopback
285/// device is present.
286#[cfg(target_os = "macos")]
287pub(crate) fn detect_macos_loopback_device() -> Option<AvfAudioDevice> {
288    let output = Command::new("ffmpeg")
289        .args([
290            "-hide_banner",
291            "-f",
292            "avfoundation",
293            "-list_devices",
294            "true",
295            "-i",
296            "",
297        ])
298        .stdin(Stdio::null())
299        .stdout(Stdio::null())
300        .stderr(Stdio::piped())
301        .output()
302        .ok()?;
303    let stderr = String::from_utf8_lossy(&output.stderr);
304    find_loopback_device(&parse_avfoundation_audio_devices(&stderr))
305}

312/// Build the `ffmpeg` argument vector for a duration-bounded `AVFoundation` audio
313/// capture to a WAV file. `audio_spec` is the `AVFoundation` audio device selector
314/// (a device index like `"1"`, or `"default"`); it is captured audio-only via
315/// the `:<spec>` input form.
316pub(crate) fn ffmpeg_avfoundation_audio_args(
317    audio_spec: &str,
318    output: &Path,
319    duration_secs: &str,
320    sample_rate_hz: u32,
321    channels: u8,
322) -> Vec<String> {
323    vec![
324        "-hide_banner".to_owned(),
325        "-nostdin".to_owned(),
326        "-loglevel".to_owned(),
327        "error".to_owned(),
328        "-f".to_owned(),
329        "avfoundation".to_owned(),
330        "-i".to_owned(),
331        format!(":{audio_spec}"),
332        "-t".to_owned(),
333        duration_secs.to_owned(),
334        "-ar".to_owned(),
335        sample_rate_hz.to_string(),
336        "-ac".to_owned(),
337        channels.to_string(),
338        "-y".to_owned(),
339        output.to_string_lossy().into_owned(),
340    ]

363fn recorders_for(platform: PlatformKind, backend: Option<AudioBackend>) -> Vec<RecorderPlan> {
364    match platform {
365        PlatformKind::MacOs => vec![
366            // ffmpeg's AVFoundation backend is the primary recorder: it can
367            // target a specific input device (system loopback via BlackHole) and
368            // is present where the legacy `afrecord` is not.
369            RecorderPlan {
370                program: "ffmpeg",
371                sample_rate_hz: 44_100,
372                channels: 2,
373                build_args: build_ffmpeg_avfoundation_args,
374            },
375            RecorderPlan {
376                program: "afrecord",
377                sample_rate_hz: 44_100,
378                channels: 1,
379                build_args: build_afrecord_args,
380            },
381        ],
382        PlatformKind::Linux => match backend {
383            Some(AudioBackend::PulseAudio) => vec![RecorderPlan {
384                program: "parecord",
385                sample_rate_hz: 48_000,
386                channels: 2,
387                build_args: build_parecord_args,
388            }],
389            // PipeWire is the preferred path; on unknown backends we still try
390            // the same recorders in preference order and rely on the runtime
391            // PATH probe to skip any that are not installed.
392            _ => vec![
393                RecorderPlan {
394                    program: "pw-record",
395                    sample_rate_hz: 48_000,
396                    channels: 2,
397                    build_args: build_pw_record_args,
398                },
399                RecorderPlan {
400                    program: "parecord",
401                    sample_rate_hz: 48_000,
402                    channels: 2,
403                    build_args: build_parecord_args,
404                },
405            ],
406        },
407        PlatformKind::Windows11 | PlatformKind::Android => Vec::new(),
408    }
409}

411fn build_pw_record_args(
412    plan: &RecorderPlan,
413    input: &ListenInput,
414    output: &Path,
415    _duration_secs: &str,
416) -> Vec<String> {
417    // pw-record writes raw PCM by default. Request a WAV container with
418    // `--format=wav` and configure rate/channels explicitly so the artifact
419    // is self-describing.
420    let target = match input.source.kind {
421        AudioSourceKind::System => "@DEFAULT_MONITOR@",
422        AudioSourceKind::Microphone | AudioSourceKind::Device => "@DEFAULT_SOURCE@",
423    };
424    // Convert the requested duration to a sample count so pw-record exits on
425    // its own: this guarantees it flushes the WAV header without us needing
426    // to send SIGTERM and race the writer.
427    let sample_count = sample_count_for(plan.sample_rate_hz, input.duration_ms);
428    vec![
429        "--target".to_owned(),
430        target.to_owned(),
431        "--rate".to_owned(),
432        plan.sample_rate_hz.to_string(),
433        "--channels".to_owned(),
434        plan.channels.to_string(),
435        "--format=s16".to_owned(),
436        "-n".to_owned(),
437        sample_count.to_string(),
438        output.to_string_lossy().into_owned(),
439    ]
440}
441
442fn sample_count_for(sample_rate_hz: u32, duration_ms: u64) -> u64 {
443    // Round up to ensure we capture at least the requested duration.
444    let numerator = u64::from(sample_rate_hz).saturating_mul(duration_ms);
445    numerator.div_ceil(1_000)
446}

491fn run_recorder(
492    plan: &RecorderPlan,
493    input: &ListenInput,
494    output: &Path,
495) -> Result<ListenArtifact, String> {
496    let duration = Duration::from_millis(input.duration_ms);
497    let duration_secs = format_seconds(duration);
498    let args = (plan.build_args)(plan, input, output, &duration_secs);
499
500    // Make sure we never inherit a stale file: caller resolves the path but
501    // does not pre-create it for recorders that prefer to allocate it
502    // themselves.
503    if output.exists() {
504        let _ = fs::remove_file(output);
505    }
506
507    let mut command = Command::new(plan.program);
508    command.args(args.iter().map(OsStr::new));
509    command
510        .stdin(Stdio::null())
511        .stdout(Stdio::null())
512        .stderr(Stdio::piped());
513
514    let mut child = command
515        .spawn()
516        .map_err(|error| format!("failed to spawn {}: {error}", plan.program))?;
517
518    if program_runs_until_killed(plan.program) {
519        wait_for_duration_then_terminate(&mut child, duration)?;
520    } else {
521        wait_with_grace(&mut child, duration)?;
522    }
523
524    // Read stderr for diagnostics regardless of success.
525    let stderr_text = drain_stderr(&mut child);
526    let status = child
527        .wait()
528        .map_err(|error| format!("failed to await {}: {error}", plan.program))?;
529
530    if !is_acceptable_exit(plan.program, status) {
531        let _ = fs::remove_file(output);
532        return Err(format!(
533            "{} exited with status {status}{}",
534            plan.program,
535            if stderr_text.trim().is_empty() {
536                String::new()
537            } else {
538                format!(": {}", stderr_text.trim())
539            }
540        ));
541    }
542
543    let metadata = fs::metadata(output).map_err(|error| {
544        format!(
545            "{} reported success but produced no file at {}: {error}",
546            plan.program,
547            output.display()
548        )
549    })?;
550    if metadata.len() == 0 {
551        let _ = fs::remove_file(output);
552        return Err(format!(
553            "{} produced an empty file at {}",
554            plan.program,
555            output.display()
556        ));
557    }
558
559    Ok(ListenArtifact {
560        path: output.to_path_buf(),
561        media_type: "audio/wav".to_owned(),
562        byte_size: metadata.len(),
563        sample_rate_hz: plan.sample_rate_hz,
564        channels: plan.channels,
565        recorder: plan.program.to_owned(),
566    })
567}

569/// Empirically, a bare WAV file with only the canonical RIFF/fmt headers and
570/// an empty data chunk weighs in at 44 bytes. Anything at or below that size
571/// means the recorder produced no PCM samples — treat those as captures with
572/// a header-only warning so callers can detect silent sessions.
573const WAV_HEADER_ONLY_BYTES: u64 = 44;
574
575/// Recorders like `parecord` keep recording until terminated. `pw-record`
576/// (with `-n COUNT`) and `afrecord` (with `-d <secs>`) exit on their own.
577fn program_runs_until_killed(program: &str) -> bool {
578    matches!(program, "parecord")
579}
580
581fn is_acceptable_exit(program: &str, status: std::process::ExitStatus) -> bool {
582    if status.success() {
583        return true;
584    }
585    // We deliberately SIGTERM long-running recorders, so a non-success exit
586    // after a timeout-kill is the normal path. Treat any exit as acceptable
587    // for those programs as long as a non-empty file was produced; the
588    // metadata check above guards against silent failures.
589    program_runs_until_killed(program)
590}
591
592fn wait_for_duration_then_terminate(child: &mut Child, duration: Duration) -> Result<(), String> {
593    let deadline = Instant::now() + duration;
594    loop {
595        if child
596            .try_wait()
597            .map_err(|error| format!("failed to poll recorder: {error}"))?
598            .is_some()
599        {
600            return Ok(());
601        }
602        if Instant::now() >= deadline {
603            // Polite shutdown first; if the recorder ignores it we escalate.
604            let _ = terminate_child(child);
605            std::thread::sleep(Duration::from_millis(150));
606            if matches!(child.try_wait(), Ok(None)) {
607                let _ = child.kill();
608            }
609            return Ok(());
610        }
611        std::thread::sleep(Duration::from_millis(25));
612    }
613}
614
615fn wait_with_grace(child: &mut Child, duration: Duration) -> Result<(), String> {
616    // Allow the recorder up to duration + 2s of headroom for container
617    // finalization. If it hangs longer we kill it so listen never blocks
618    // an agent indefinitely.
619    let deadline = Instant::now() + duration + Duration::from_secs(2);
620    loop {
621        if child
622            .try_wait()
623            .map_err(|error| format!("failed to poll recorder: {error}"))?
624            .is_some()
625        {
626            return Ok(());
627        }
628        if Instant::now() >= deadline {
629            let _ = child.kill();
630            return Ok(());
631        }
632        std::thread::sleep(Duration::from_millis(25));
633    }
634}

636#[cfg(unix)]
637fn terminate_child(child: &Child) -> std::io::Result<()> {
638    // Send SIGTERM by shelling out to /bin/kill. We deliberately avoid
639    // libc::kill here because the workspace forbids `unsafe_code`. Using a
640    // SIGTERM (instead of std::process::Child::kill which sends SIGKILL) lets
641    // recorders flush their WAV headers before exiting.
642    let pid = child.id();
643    let status = std::process::Command::new("kill")
644        .arg("-TERM")
645        .arg(pid.to_string())
646        .stdin(Stdio::null())
647        .stdout(Stdio::null())
648        .stderr(Stdio::null())
649        .status()?;
650    if status.success() {
651        Ok(())
652    } else {
653        Err(std::io::Error::other(format!(
654            "kill -TERM {pid} exited with status {status}"
655        )))
656    }
657}

659#[cfg(not(unix))]
660fn terminate_child(child: &Child) -> std::io::Result<()> {
661    // On non-unix platforms we have no graceful signal; fall back to kill().
662    // The current code path only invokes this from the Linux capture branch.
663    let mut handle = child.stdin.as_ref();
664    let _ = handle.take();
665    Ok(())
666}
667
668fn drain_stderr(child: &mut Child) -> String {
669    let mut buffer = String::new();
670    if let Some(mut stderr) = child.stderr.take() {
671        let _ = stderr.read_to_string(&mut buffer);
672    }
673    buffer
674}

676fn command_available(program: &str) -> bool {
677    // Probe via PATH walking ourselves, so we avoid pulling in the `which`
678    // crate just for this check.
679    let Some(path) = std::env::var_os("PATH") else {
680        return false;
681    };
682    for entry in std::env::split_paths(&path) {
683        let candidate = entry.join(program);
684        if candidate.is_file() {
685            #[cfg(unix)]
686            {
687                use std::os::unix::fs::PermissionsExt;
688                if let Ok(metadata) = candidate.metadata() {
689                    if metadata.permissions().mode() & 0o111 != 0 {
690                        return true;
691                    }
692                }
693            }
694            #[cfg(not(unix))]
695            {
696                return true;
697            }
698        }
699    }
700    false
701}

703fn resolve_output_path(output: Option<&Path>) -> Result<PathBuf, TendrilError> {
704    if let Some(path) = output {
705        if let Some(parent) = path.parent() {
706            if !parent.as_os_str().is_empty() && !parent.exists() {
707                return Err(TendrilError::validation(format!(
708                    "listen --output parent directory `{}` does not exist",
709                    parent.display()
710                ))
711                .with_code("invalid_listen_input")
712                .with_field("output"));
713            }
714        }
715        return Ok(path.to_path_buf());
716    }
717
718    let mut candidate = std::env::temp_dir();
719    let unique = format!(
720        "tendril-listen-{}-{}.wav",
721        std::process::id(),
722        unique_suffix()
723    );
724    candidate.push(unique);
725    Ok(candidate)
726}
727
728fn unique_suffix() -> String {
729    use std::time::{SystemTime, UNIX_EPOCH};
730    SystemTime::now().duration_since(UNIX_EPOCH).map_or_else(
731        |_| "0".to_owned(),
732        |duration| duration.as_nanos().to_string(),
733    )
734}
735
736fn format_seconds(duration: Duration) -> String {
737    // afrecord accepts fractional seconds; clamp to millisecond precision.
738    let total_millis = duration.as_millis();
739    if total_millis % 1_000 == 0 {
740        format!("{}", total_millis / 1_000)
741    } else {
742        format!("{:.3}", duration.as_secs_f64())
743    }
744}
