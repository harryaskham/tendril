38pub fn discover_elements(
39    adapter: &AdapterInfo,
40    inventory: &TargetInventory,
41    input: &ElementListInput,
42) -> Result<ElementListOutput, TendrilError> {
43    input.validate()?;
44
45    if let Some(fixture) = load_fixture(adapter, input)? {
46        return Ok(fixture);
47    }
48
49    let targets = matching_targets(inventory, input)?;
50    let mut notes = Vec::new();
51    let mut elements = match (adapter.platform, adapter.session) {
52        (PlatformKind::MacOs, DesktopSession::MacOsWindowServer) => {
53            discover_macos_accessibility_elements(
54                inventory,
55                &targets,
56                input.include_offscreen,
57                &mut notes,
58            )
59        }
60        (PlatformKind::Linux, DesktopSession::X11) => {
61            discover_x11_elements(&targets, input.include_offscreen, &mut notes)
62        }
63        (PlatformKind::Linux, DesktopSession::Wayland) => {
64            discover_wayland_elements(&targets, input.include_offscreen, &mut notes)
65        }
66        (PlatformKind::Windows11, DesktopSession::WindowsDesktop) => {
67            discover_windows_elements(inventory, &targets, input.include_offscreen, &mut notes)
68        }
69        _ => {
70            notes.push(
71                "No platform element backend is available for this session; returning target roots only."
72                    .to_owned(),
73            );
74            Vec::new()
75        }
76    };
77
78    if elements.is_empty() {
79        notes.push("No child UI elements were reported; returning target roots.".to_owned());
80        elements = root_elements_from_targets(&targets);
81    }
82
83    assign_snapshot_ids(&mut elements);
84    Ok(ElementListOutput {
85        adapter: adapter.clone(),
86        target: input.target.clone(),
87        elements,
88        notes,
89    })
90}

122fn matching_targets(
123    inventory: &TargetInventory,
124    input: &ElementListInput,
125) -> Result<Vec<PlatformTargetDescriptor>, TendrilError> {
126    let targets = inventory
127        .targets
128        .iter()
129        .filter(|target| match &input.target {
130            None => true,
131            Some(selector) => {
132                target.id == selector.id() && selector_matches_kind(selector, target.kind)
133            }
134        })
135        .cloned()
136        .collect::<Vec<_>>();
137
138    if targets.is_empty() {
139        if let Some(selector) = &input.target {
140            return Err(TendrilError::target_not_found(
141                match selector.kind() {
142                    TargetKind::Window => "window",
143                    TargetKind::Display => "display",
144                    TargetKind::AudioSource => "audio_source",
145                },
146                selector.id(),
147            ));
148        }
149    }
150
151    Ok(targets)
152}

157        (TargetSelector::Window { .. }, CaptureTargetKind::Window)

159    )

205fn discover_macos_accessibility_elements(
206    inventory: &TargetInventory,
207    targets: &[PlatformTargetDescriptor],
208    include_offscreen: bool,
209    notes: &mut Vec<String>,
210) -> Vec<ElementDescriptor> {
211    let mut elements = Vec::new();
212    let query_targets = macos_accessibility_query_targets(inventory, targets, notes);
213    for target in &query_targets {
214        let Some(process_id) = target.process_id else {
215            elements.push(root_element_from_target(target));
216            notes.push(format!(
217                "Target `{}` did not expose a process id, so macOS Accessibility elements could not be queried.",
218                target.id
219            ));
220            continue;
221        };
222
223        match run_macos_accessibility_listing(process_id, target, include_offscreen) {
224            Ok(mut discovered) => elements.append(&mut discovered),
225            Err(error) => {
226                notes.push(format!(
227                    "macOS Accessibility element listing failed for `{}`: {error}; returning its root target instead.",
228                    target.id
229                ));
230                elements.push(root_element_from_target(target));
231            }
232        }
233    }
234    elements

237fn macos_accessibility_query_targets(
238    inventory: &TargetInventory,
239    targets: &[PlatformTargetDescriptor],
240    notes: &mut Vec<String>,
241) -> Vec<PlatformTargetDescriptor> {
242    let mut query_targets = Vec::new();
243    let mut seen = HashSet::new();
244
245    for target in targets {
246        match target.kind {
247            CaptureTargetKind::Window => push_unique_target(&mut query_targets, &mut seen, target),
248            CaptureTargetKind::Display => {
249                let before = query_targets.len();
250                for window in inventory.targets.iter().filter(|candidate| {
251                    candidate.kind == CaptureTargetKind::Window
252                        && bounds_overlap(&candidate.bounds, &target.bounds)
253                }) {
254                    push_unique_target(&mut query_targets, &mut seen, window);
255                }
256                if query_targets.len() == before {
257                    notes.push(format!(
258                        "Display `{}` did not contain any discovered windows for macOS Accessibility listing; returning its root target instead.",
259                        target.id
260                    ));
261                    push_unique_target(&mut query_targets, &mut seen, target);
262                }
263            }
264        }
265    }
266
267    query_targets
268}

270fn push_unique_target(
271    targets: &mut Vec<PlatformTargetDescriptor>,
272    seen: &mut HashSet<String>,
273    target: &PlatformTargetDescriptor,
274) {
275    let key = format!("{:?}:{}", target.kind, target.id);

277        targets.push(target.clone());
278    }

281fn run_macos_accessibility_listing(
282    process_id: u32,
283    target: &PlatformTargetDescriptor,
284    include_offscreen: bool,
285) -> Result<Vec<ElementDescriptor>, String> {
286    let script = macos_accessibility_listing_jxa(process_id, target, include_offscreen);
287    let output = Command::new("osascript")
288        .args(["-l", "JavaScript", "-e", &script])
289        .output()
290        .map_err(|error| format!("failed to spawn osascript: {error}"))?;
291    if !output.status.success() {
292        let stderr = String::from_utf8_lossy(&output.stderr).trim().to_owned();
293        let stdout = String::from_utf8_lossy(&output.stdout).trim().to_owned();
294        return Err(if stdout.is_empty() || stderr.is_empty() {
295            format!("{stdout}{stderr}")
296        } else {
297            format!("{stdout} | {stderr}")
298        });
299    }
300    let stdout = String::from_utf8_lossy(&output.stdout);
301    let values = serde_json::from_str::<Value>(stdout.trim())
302        .map_err(|error| format!("failed to parse osascript JSON: {error}"))?;
303    let mut elements = Vec::new();
304    if let Some(array) = values.as_array() {
305        for value in array {
306            let role = json_str(value, "role").unwrap_or("element").to_owned();
307            let name = json_str(value, "name")
308                .or_else(|| json_str(value, "description"))
309                .or_else(|| json_str(value, "value"))
310                .unwrap_or(&role)
311                .to_owned();
312            let bounds = value.get("bounds").map(|bounds| Bounds {
313                x: json_i32(bounds, "x").unwrap_or(target.bounds.x),
314                y: json_i32(bounds, "y").unwrap_or(target.bounds.y),
315                width: json_u32(bounds, "width").unwrap_or(1),
316                height: json_u32(bounds, "height").unwrap_or(1),
317            });
318            elements.push(ElementDescriptor {
319                id: String::new(),
320                role,
321                name,
322                description: json_str(value, "description").map(str::to_owned),
323                value: json_str(value, "value").map(str::to_owned),
324                bounds,
325                target: Some(target_selector_from_platform(target)),
326                path: value.get("path").and_then(Value::as_array).map_or_else(
327                    || target.app_name.iter().cloned().collect(),
328                    |items| {
329                        items
330                            .iter()
331                            .filter_map(Value::as_str)
332                            .map(str::to_owned)
333                            .collect()
334                    },
335                ),
336                actions: vec!["click".to_owned(), "press".to_owned()],
337                app_name: target.app_name.clone(),
338                process_id: target.process_id,
339            });
340        }
341    }
342    Ok(elements)
343}

345fn macos_accessibility_listing_jxa(
346    process_id: u32,
347    target: &PlatformTargetDescriptor,
348    include_offscreen: bool,
349) -> String {
350    let target_x = target.bounds.x;
351    let target_y = target.bounds.y;
352    let target_width = target.bounds.width;
353    let target_height = target.bounds.height;
354    let include_offscreen = if include_offscreen { "true" } else { "false" };
355    format!(
356        r"ObjC.import('ApplicationServices');
357(function () {{
358  var pid = {process_id};
359  var target = {{ x: {target_x}, y: {target_y}, width: {target_width}, height: {target_height} }};
360  var includeOffscreen = {include_offscreen};
361  var maxElements = {MAX_MACOS_ELEMENTS};
362  var app = $.AXUIElementCreateApplication(pid);
363  if (!app) throw new Error('AXUIElementCreateApplication failed');
364  var out = [];
365  function unwrap(value) {{
366    try {{ return ObjC.unwrap(value); }} catch (e) {{ return String(value); }}
367  }}
368  function attr(element, name) {{
369    var ref = Ref();
370    var err = $.AXUIElementCopyAttributeValue(element, $(name), ref);
371    if (err !== 0 || !ref[0]) return null;
372    return ObjC.castRefToObject(ref[0]);
373  }}
374  function point(value) {{
375    if (!value) return null;
376    var p = $.CGPointMake(0, 0);
377    if (!$.AXValueGetValue(value, 1, Ref(p))) return null;
378    return {{ x: Math.round(p.x), y: Math.round(p.y) }};
379  }}
380  function size(value) {{
381    if (!value) return null;
382    var s = $.CGSizeMake(0, 0);
383    if (!$.AXValueGetValue(value, 2, Ref(s))) return null;
384    return {{ width: Math.round(s.width), height: Math.round(s.height) }};
385  }}
386  function intersects(bounds) {{
387    if (includeOffscreen || !bounds) return true;
388    return bounds.x < target.x + target.width && bounds.x + bounds.width > target.x &&
389           bounds.y < target.y + target.height && bounds.y + bounds.height > target.y;
390  }}
391  function textAttr(element, name) {{
392    var value = attr(element, name);
393    if (!value) return null;
394    var text = String(unwrap(value));
395    return text.length === 0 ? null : text;
396  }}
397  function roleName(role) {{
398    if (!role) return 'element';
399    return String(unwrap(role)).replace(/^AX/, '').toLowerCase();
400  }}
401  function walk(element, depth, path) {{
402    if (out.length >= maxElements || depth > 8) return;
403    var role = roleName(attr(element, 'AXRole'));
404    var title = textAttr(element, 'AXTitle');
405    var desc = textAttr(element, 'AXDescription');
406    var value = textAttr(element, 'AXValue');
407    var pos = point(attr(element, 'AXPosition'));
408    var sz = size(attr(element, 'AXSize'));
409    var bounds = (pos && sz) ? {{ x: pos.x, y: pos.y, width: sz.width, height: sz.height }} : null;
410    var name = title || desc || value || role;
411    if (intersects(bounds)) {{
412      out.push({{ role: role, name: name, description: desc, value: value, bounds: bounds, path: path.concat([name]) }});
413    }}
414    var children = attr(element, 'AXChildren');
415    if (!children) return;
416    var count = children.count;
417    for (var i = 0; i < count && out.length < maxElements; i += 1) {{
418      walk(children.objectAtIndex(i), depth + 1, path.concat([name]));
419    }}
420  }}
421  walk(app, 0, []);
422  return JSON.stringify(out);
423}}());
424"
425    )
426}

457fn discover_x11_window_elements(
458    targets: &[PlatformTargetDescriptor],
459    include_offscreen: bool,
460    notes: &mut Vec<String>,
461) -> Vec<ElementDescriptor> {
462    let mut elements = Vec::new();
463    for target in targets {
464        if target.kind != CaptureTargetKind::Window || !program_on_path("xwininfo") {
465            elements.push(root_element_from_target(target));
466            if target.kind == CaptureTargetKind::Window {
467                notes.push(
468                    "xwininfo was not available; returning X11 target roots only for element listing."
469                        .to_owned(),
470                );
471            }
472            continue;
473        }
474        match Command::new("xwininfo")
475            .args(["-tree", "-id", &target.id])
476            .output()
477        {
478            Ok(output) if output.status.success() => {
479                let stdout = String::from_utf8_lossy(&output.stdout);
480                let mut discovered = parse_xwininfo_tree(&stdout, target, include_offscreen);
481                if discovered.is_empty() {
482                    elements.push(root_element_from_target(target));
483                } else {
484                    elements.append(&mut discovered);
485                }
486            }
487            Ok(output) => {
488                notes.push(format!(
489                    "xwininfo failed for `{}`: {}; returning its root target.",
490                    target.id,
491                    String::from_utf8_lossy(&output.stderr).trim()
492                ));
493                elements.push(root_element_from_target(target));
494            }
495            Err(error) => {
496                notes.push(format!(
497                    "failed to spawn xwininfo for `{}`: {error}; returning its root target.",
498                    target.id
499                ));
500                elements.push(root_element_from_target(target));
501            }
502        }
503    }
504    elements
505}

518fn parse_xwininfo_line(
519    line: &str,
520    target: &PlatformTargetDescriptor,
521    include_offscreen: bool,
522) -> Option<ElementDescriptor> {
523    let trimmed = line.trim_start();
524    if !trimmed.starts_with("0x") {
525        return None;
526    }
527    let xid = trimmed.split_whitespace().next()?.to_owned();
528    let quoted_name = quoted_segment(trimmed).unwrap_or_else(|| xid.clone());
529    let (bounds, _) = parse_x11_geometry_from_line(trimmed, &target.bounds)?;
530    if !include_offscreen && !bounds_overlap(&bounds, &target.bounds) {
531        return None;
532    }
533    Some(ElementDescriptor {
534        id: String::new(),
535        role: if xid == target.id {
536            "window"
537        } else {
538            "x11_window"
539        }
540        .to_owned(),
541        name: quoted_name,
542        description: Some(format!("X11 window {xid}")),
543        value: None,
544        bounds: Some(bounds),
545        target: Some(target_selector_from_platform(target)),
546        path: target
547            .app_name
548            .iter()
549            .chain(std::iter::once(&target.name))
550            .cloned()
551            .collect(),
552        actions: vec!["click".to_owned()],
553        app_name: target.app_name.clone(),
554        process_id: target.process_id,
555    })
556}

558fn quoted_segment(line: &str) -> Option<String> {
559    let start = line.find('"')? + 1;
560    let end = line[start..].find('"')? + start;
561    Some(line[start..end].to_owned())
562}

564fn parse_x11_geometry_from_line(line: &str, target: &Bounds) -> Option<(Bounds, bool)> {
565    let tokens = line.split_whitespace().collect::<Vec<_>>();
566    let geometry = tokens.iter().find_map(|token| parse_size_offset(token))?;
567    let absolute_offset = tokens.iter().find_map(|token| parse_offset_only(token));
568    let (width, height, rel_x, rel_y) = geometry;
569    let (x, y, absolute) = if let Some((abs_x, abs_y)) = absolute_offset {
570        (abs_x, abs_y, true)
571    } else {
572        (
573            target.x.saturating_add(rel_x),
574            target.y.saturating_add(rel_y),
575            false,
576        )
577    };
578    Some((
579        Bounds {
580            x,
581            y,
582            width,
583            height,
584        },
585        absolute,
586    ))
587}

589fn parse_size_offset(token: &str) -> Option<(u32, u32, i32, i32)> {
590    let (width_text, rest) = token.split_once('x')?;
591    let (height_text, x_text, y_text) = split_geometry_offsets(rest)?;
592    Some((
593        width_text.parse().ok()?,
594        height_text.parse().ok()?,
595        x_text.parse().ok()?,
596        y_text.parse().ok()?,
597    ))
598}
599
600fn parse_offset_only(token: &str) -> Option<(i32, i32)> {
601    if token.contains('x') {
602        return None;
603    }
604    let (_, x_text, y_text) = split_geometry_offsets(token)?;
605    Some((x_text.parse().ok()?, y_text.parse().ok()?))
606}

608fn split_geometry_offsets(value: &str) -> Option<(&str, &str, &str)> {
609    let sign_index = value
610        .char_indices()
611        .skip(1)
612        .find_map(|(index, ch)| matches!(ch, '+' | '-').then_some(index))?;
613    let first = &value[..sign_index];
614    let rest = &value[sign_index..];
615    let second_sign = rest
616        .char_indices()
617        .skip(1)
618        .find_map(|(index, ch)| matches!(ch, '+' | '-').then_some(index))?;
619    Some((first, &rest[..second_sign], &rest[second_sign..]))
620}

622fn discover_windows_elements(
623    inventory: &TargetInventory,
624    targets: &[PlatformTargetDescriptor],
625    include_offscreen: bool,
626    notes: &mut Vec<String>,
627) -> Vec<ElementDescriptor> {
628    let query_targets = window_targets_for_scope(inventory, targets);
629    let mut elements = Vec::new();
630    for target in &query_targets {
631        if target.kind != CaptureTargetKind::Window {
632            elements.push(root_element_from_target(target));
633            continue;
634        }
635        match tendril_win32::discover_window_elements(&target.id) {
636            Ok(discovered) if !discovered.is_empty() => {
637                elements.extend(discovered.into_iter().filter_map(|element| {
638                    windows_element_descriptor(target, element, include_offscreen)
639                }));
640            }
641            Ok(_) => elements.push(root_element_from_target(target)),
642            Err(error) => {
643                notes.push(format!(
644                    "Windows native element listing failed for `{}`: {error}; returning its root target instead.",
645                    target.id
646                ));
647                elements.push(root_element_from_target(target));
648            }
649        }
650    }
651    if !elements.is_empty() {
652        notes.push(
653            "Windows element discovery used native Win32 window/control enumeration; element bounds are screen coordinates and click(<id>) resolves them through the shared target-relative DSL contract."
654                .to_owned(),
655        );
656    }
657    elements
658}

660fn window_targets_for_scope(

662    targets: &[PlatformTargetDescriptor],

665    let mut seen = HashSet::new();

678    }

681    } else {

683    }

866#[cfg(target_os = "linux")]
867fn run_atspi_accessibility_listing(
868    targets: &[PlatformTargetDescriptor],
869    include_offscreen: bool,
870) -> Result<Vec<ElementDescriptor>, String> {
871    let client = AtspiClient::connect()?;
872    let applications = client.applications()?;
873    let mut elements = Vec::new();
874
875    for target in targets {
876        for application in &applications {
877            if elements.len() >= MAX_ATSPI_ELEMENTS {
878                return Ok(elements);
879            }
880            let process_id = atspi_application_process_id(&client, application);
881            if target.kind == CaptureTargetKind::Window
882                && target.process_id.is_some()
883                && process_id.is_some()
884                && target.process_id != process_id
885            {
886                continue;
887            }
888            let app_name = atspi_accessible_name(&client, application)
889                .filter(|name| !name.trim().is_empty())
890                .or_else(|| target.app_name.clone());
891            let mut visited = HashSet::new();
892            let root_path = app_name.clone().into_iter().collect::<Vec<_>>();
893            walk_atspi_tree(
894                &client,
895                application,
896                target,
897                include_offscreen,
898                &root_path,
899                app_name.as_ref(),
900                process_id.or(target.process_id),
901                &mut visited,
902                &mut elements,
903                0,
904            );
905        }
906    }
907
908    Ok(elements)
909}

911#[allow(clippy::too_many_arguments)]
912#[cfg(target_os = "linux")]
913fn walk_atspi_tree(
914    client: &AtspiClient,
915    object: &AtspiObjectRef,
916    target: &PlatformTargetDescriptor,
917    include_offscreen: bool,
918    parent_path: &[String],
919    app_name: Option<&String>,
920    process_id: Option<u32>,
921    visited: &mut HashSet<String>,
922    elements: &mut Vec<ElementDescriptor>,
923    depth: usize,
924) {
925    if elements.len() >= MAX_ATSPI_ELEMENTS || depth > MAX_ATSPI_DEPTH {
926        return;
927    }
928    if !visited.insert(object.key()) {
929        return;
930    }
931
932    let Ok(accessible) = client.accessible_proxy(object) else {
933        return;
934    };
935    let role = atspi_role_name(&accessible);
936    let name = atspi_string_property(&accessible, "Name")
937        .or_else(|| atspi_string_property(&accessible, "Description"))
938        .unwrap_or_else(|| role.clone());
939    let description = atspi_string_property(&accessible, "Description");
940    let value = atspi_value_text(client, object);
941    let bounds = atspi_component_extents(client, object);
942    let mut path = parent_path.to_vec();
943    if path.last() != Some(&name) {
944        path.push(name.clone());
945    }
946
947    if atspi_element_in_scope(bounds.as_ref(), &target.bounds, include_offscreen) {
948        let mut actions = atspi_action_names(client, object);
949        if bounds.is_some() && !actions.iter().any(|action| action == "click") {
950            actions.insert(0, "click".to_owned());
951        }
952        if bounds.is_some() && !actions.iter().any(|action| action == "press") {
953            actions.push("press".to_owned());
954        }
955        elements.push(ElementDescriptor {
956            id: String::new(),
957            role,
958            name: name.clone(),
959            description,
960            value,
961            bounds,
962            target: Some(target_selector_from_platform(target)),
963            path,
964            actions,
965            app_name: app_name.cloned(),
966            process_id,
967        });
968    }
969
970    for child in atspi_children(&accessible) {
971        let child_parent_path = parent_path_for_child(parent_path, &name);
972        walk_atspi_tree(
973            client,
974            &child,
975            target,
976            include_offscreen,
977            &child_parent_path,
978            app_name,
979            process_id,
980            visited,
981            elements,
982            depth + 1,
983        );
984    }
985}

1153fn bounds_overlap(left: &Bounds, right: &Bounds) -> bool {
1154    let left_right = i64::from(left.x) + i64::from(left.width);
1155    let left_bottom = i64::from(left.y) + i64::from(left.height);
1156    let right_right = i64::from(right.x) + i64::from(right.width);
1157    let right_bottom = i64::from(right.y) + i64::from(right.height);
1158    i64::from(left.x) < right_right
1159        && left_right > i64::from(right.x)
1160        && i64::from(left.y) < right_bottom
1161        && left_bottom > i64::from(right.y)
1162}
